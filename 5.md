The improved design should use a two-stage synchronization protocol rather than PREPARE → GO:

CONTROL CHANNEL

1. SYNC exchanges
   reference master ↔ test master

2. PREPARE
   channel + duration + TX/RX roles + test ID

3. READY
   both masters and both Serial1 slaves armed

4. START_AT(T)
   absolute start time expressed in each master's local µs clock

5. RF WINDOW
   ─────────────────────────────────────────────
   NO ESP-NOW
   NO Serial1 traffic
   fixed channel
   fixed TX rate
   fixed packet interval
   CSI + packet filtering
   ─────────────────────────────────────────────

6. RETURN_TO_CONTROL

7. RESULT

This is significantly better because ESP-NOW's channel must match the peer's channel, and Espressif explicitly notes that ESP-NOW delivery can fail when the devices are on different channels.  CSI should remain in promiscuous mode, with the CSI callback kept extremely short because it executes in the Wi-Fi task. 

For the test-node master, I would therefore replace the previous implementation with the following core.

// ============================================================
// CSI CALLBACK
// ============================================================

static void IRAM_ATTR csiCallback(
    void *ctx,
    wifi_csi_info_t *data)
{
    if (data == nullptr) {
        return;
    }

    if (data->buf == nullptr) {
        return;
    }

    if (data->len < 2) {
        return;
    }

    const int pairs =
        min(
            data->len / 2,
            static_cast<int>(
                MAX_CSI_PAIRS
            )
        );

    const int8_t rssi =
        data->rx_ctrl.rssi;


    portENTER_CRITICAL_ISR(
        &gCSIMux
    );

    gCSI.frames++;

    gCSI.rssiSum += rssi;

    gCSI.rssiSumSq +=
        static_cast<uint32_t>(
            rssi * rssi
        );

    if (rssi < gCSI.rssiMin) {
        gCSI.rssiMin = rssi;
    }

    if (rssi > gCSI.rssiMax) {
        gCSI.rssiMax = rssi;
    }

    gCSI.pairCountTotal += pairs;


    for (int i = 0;
         i < pairs;
         ++i) {

        /*
         * Classic ESP32 CSI data consists of signed I/Q
         * byte pairs. The exact ordering must be validated
         * against the installed Arduino-ESP32/IDF revision
         * before assigning absolute frequency labels.
         */

        const int8_t imag =
            data->buf[
                2 * i
            ];

        const int8_t real =
            data->buf[
                2 * i + 1
            ];


        gCSI.sumI[i] += real;

        gCSI.sumQ[i] += imag;

        gCSI.count[i]++;
    }


    portEXIT_CRITICAL_ISR(
        &gCSIMux
    );
}


// ============================================================
// PROMISCUOUS PACKET FILTER
// ============================================================
//
// We want:
//   management frames
//   data frames
//
// and subsequently identify OUR test packets ourselves.
//
// ============================================================

static void promiscuousCallback(
    void *packet,
    wifi_promiscuous_pkt_type_t type)
{
    if (packet == nullptr) {
        return;
    }

    if (type != WIFI_PKT_MGMT &&
        type != WIFI_PKT_DATA) {
        return;
    }

    wifi_promiscuous_pkt_t *pkt =
        static_cast<
            wifi_promiscuous_pkt_t *
        >(packet);


    const uint16_t len =
        pkt->rx_ctrl.sig_len;


    if (len < 32) {
        return;
    }


    const uint8_t *frame =
        pkt->payload;


    /*
     * Frame control:
     *
     * bits 2..3 = type
     * subtype for management action = 13
     */

    const uint16_t fc =
        frame[0] |
        (static_cast<uint16_t>(
            frame[1]
        ) << 8);


    const uint8_t frameType =
        (fc >> 2) & 0x03;

    const uint8_t subtype =
        (fc >> 4) & 0x0F;


    // Our raw test frame is a management ACTION frame.

    if (frameType != 0 ||
        subtype != 13) {

        return;
    }


    /*
     * We use:
     *
     *   category = 127
     *   OUI      = DE:AD:BE
     *   type     = 01
     *   magic    = RFTS
     *
     * Offset:
     *
     * FC         0..1
     * duration   2..3
     * DA         4..9
     * SA         10..15
     * BSSID      16..21
     * seq        22..23
     * category   24
     * OUI        25..27
     * subtype    28
     * magic      29..32
     */

    if (len < 40) {
        return;
    }


    const uint8_t category =
        frame[24];

    if (category != 127) {
        return;
    }


    if (frame[25] != 0xDE ||
        frame[26] != 0xAD ||
        frame[27] != 0xBE) {

        return;
    }


    if (frame[28] != 0x01) {
        return;
    }


    if (frame[29] != 0x52 ||
        frame[30] != 0x46 ||
        frame[31] != 0x54 ||
        frame[32] != 0x53) {

        return;
    }


    ++gPacketStats.matchingPackets;


    if (len >= 42) {

        uint32_t sequence =
            frame[39] |
            (static_cast<uint32_t>(
                frame[40]
            ) << 8) |
            (static_cast<uint32_t>(
                frame[41]
            ) << 16) |
            (static_cast<uint32_t>(
                frame[42]
            ) << 24);


        if (gPacketStats.haveSequence) {

            if (sequence <=
                gPacketStats.lastSequence) {

                ++gPacketStats.duplicates;
            }
        }


        gPacketStats.lastSequence =
            sequence;

        gPacketStats.haveSequence = true;
    }


    ++gPacketStats.packets;
}

static void initWiFi()
{
    WiFi.mode(
        WIFI_STA
    );

    WiFi.disconnect(
        true,
        true
    );

    delay(50);


    esp_err_t err;


    err =
        esp_wifi_set_ps(
            WIFI_PS_NONE
        );


    if (err != ESP_OK) {

        Serial.printf(
            "WARNING: power save: %s\n",
            esp_err_to_name(err)
        );
    }


    err =
        esp_wifi_set_channel(
            CONTROL_CHANNEL,
            WIFI_SECOND_CHAN_NONE
        );


    if (err != ESP_OK) {

        Serial.printf(
            "FATAL channel: %s\n",
            esp_err_to_name(err)
        );

        while (true) {
            delay(1000);
        }
    }


    /*
     * Promiscuous mode is needed for CSI when we are not
     * associated as a station.
     */

    err =
        esp_wifi_set_promiscuous(
            true
        );


    if (err != ESP_OK) {

        Serial.printf(
            "FATAL promiscuous: %s\n",
            esp_err_to_name(err)
        );

        while (true) {
            delay(1000);
        }
    }


    /*
     * Separate promiscuous callback lets us reject unrelated
     * frames at the application level.
     */

    err =
        esp_wifi_set_promiscuous_rx_cb(
            promiscuousCallback
        );


    if (err != ESP_OK) {

        Serial.printf(
            "FATAL promisc callback: %s\n",
            esp_err_to_name(err)
        );

        while (true) {
            delay(1000);
        }
    }


    /*
     * CSI callback.
     */

    err =
        esp_wifi_set_csi_rx_cb(
            csiCallback,
            nullptr
        );


    if (err != ESP_OK) {

        Serial.printf(
            "FATAL CSI callback: %s\n",
            esp_err_to_name(err)
        );

        while (true) {
            delay(1000);
        }
    }


    wifi_csi_config_t config{};

    config.lltf_en = true;

    config.htltf_en = true;

    config.stbc_htltf2_en = true;

    config.ltf_merge_en = false;

    config.channel_filter_en = false;

    config.manu_scale = false;

    config.shift = 0;


    err =
        esp_wifi_set_csi_config(
            &config
        );


    if (err != ESP_OK) {

        Serial.printf(
            "FATAL CSI config: %s\n",
            esp_err_to_name(err)
        );

        while (true) {
            delay(1000);
        }
    }


    err =
        esp_wifi_set_csi(true);


    if (err != ESP_OK) {

        Serial.printf(
            "FATAL CSI enable: %s\n",
            esp_err_to_name(err)
        );

        while (true) {
            delay(1000);
        }
    }


    /*
     * Fixed maximum TX power for repeatability.
     *
     * Actual EIRP depends on antenna gain and board.
     */

    err =
        esp_wifi_set_max_tx_power(
            78
        );


    if (err != ESP_OK) {

        Serial.printf(
            "WARNING TX power: %s\n",
            esp_err_to_name(err)
        );
    }
}


// ============================================================
// ESP-NOW CALLBACK
// ============================================================

static void espNowReceiveCallback(
    const esp_now_recv_info_t *info,
    const uint8_t *data,
    int len)
{
    if (info == nullptr ||
        data == nullptr) {
        return;
    }


    if (memcmp(
            info->src_addr,
            REFERENCE_MASTER_MAC,
            6
        ) != 0) {

        return;
    }


    if (len <= 0 ||
        len > 250) {

        return;
    }


    EspNowMessage msg{};

    memcpy(
        msg.mac,
        info->src_addr,
        6
    );


    msg.length =
        static_cast<uint16_t>(
            len
        );


    memcpy(
        msg.data,
        data,
        len
    );


    if (gEspNowQueue) {

        xQueueSend(
            gEspNowQueue,
            &msg,
            0
        );
    }
}


// ============================================================
// ESP-NOW INIT
// ============================================================

static void initEspNow()
{
    gEspNowQueue =
        xQueueCreate(
            8,
            sizeof(EspNowMessage)
        );


    if (!gEspNowQueue) {

        Serial.println(
            "FATAL: ESP-NOW queue"
        );

        while (true) {
            delay(1000);
        }
    }


    esp_err_t err =
        esp_now_init();


    if (err != ESP_OK) {

        Serial.printf(
            "FATAL esp_now_init: %s\n",
            esp_err_to_name(err)
        );

        while (true) {
            delay(1000);
        }
    }


    esp_now_register_recv_cb(
        espNowReceiveCallback
    );


    esp_now_peer_info_t peer{};

    memcpy(
        peer.peer_addr,
        REFERENCE_MASTER_MAC,
        6
    );


    peer.channel =
        CONTROL_CHANNEL;


    peer.ifidx =
        WIFI_IF_STA;


    peer.encrypt = false;


    if (!esp_now_is_peer_exist(
            REFERENCE_MASTER_MAC
        )) {

        err =
            esp_now_add_peer(
                &peer
            );


        if (err != ESP_OK) {

            Serial.printf(
                "FATAL peer: %s\n",
                esp_err_to_name(err)
            );

            while (true) {
                delay(1000);
            }
        }
    }
}

// ============================================================
// CLOCK SYNC
// ============================================================

static bool performClockSync()
{
    constexpr uint8_t ITERATIONS = 8;


    int64_t bestOffset = 0;

    uint64_t bestDelay =
        UINT64_MAX;


    for (uint8_t i = 0;
         i < ITERATIONS;
         ++i) {


        SyncRequest req{};

        req.magic =
            RF_MAGIC;

        req.version =
            PROTOCOL_VERSION;

        req.type =
            static_cast<uint8_t>(
                ControlType::SYNC_REQ
            );

        req.sequence =
            i;


        req.t1 =
            esp_timer_get_time();


        gEspNowQueue =
            gEspNowQueue;


        if (!sendControl(
                &req,
                sizeof(req)
            )) {

            continue;
        }


        const uint64_t waitStart =
            esp_timer_get_time();


        while (
            esp_timer_get_time() -
            waitStart <
            100000
        ) {


            EspNowMessage msg{};


            if (
                xQueueReceive(
                    gEspNowQueue,
                    &msg,
                    pdMS_TO_TICKS(2)
                ) != pdTRUE
            ) {

                continue;
            }


            if (
                msg.length <
                sizeof(
                    SyncResponse
                )
            ) {

                continue;
            }


            const SyncResponse *resp =
                reinterpret_cast<
                    const SyncResponse *
                >(msg.data);


            if (
                resp->magic !=
                    RF_MAGIC ||
                resp->version !=
                    PROTOCOL_VERSION ||
                resp->type !=
                    static_cast<uint8_t>(
                        ControlType::SYNC_RESP
                    ) ||
                resp->sequence !=
                    i
            ) {

                continue;
            }


            const uint64_t t4 =
                esp_timer_get_time();


            const int64_t offset =
                (
                    static_cast<
                        int64_t
                    >(resp->t2)
                    -
                    static_cast<
                        int64_t
                    >(req.t1)
                    +
                    static_cast<
                        int64_t
                    >(resp->t3)
                    -
                    static_cast<
                        int64_t
                    >(t4)
                ) / 2;


            const uint64_t delay =
                (
                    t4 -
                    req.t1
                ) -
                (
                    resp->t3 -
                    resp->t2
                );


            if (
                delay <
                bestDelay
            ) {

                bestDelay = delay;

                bestOffset = offset;
            }


            break;
        }


        delay(5);
    }


    if (bestDelay ==
        UINT64_MAX) {

        gClockSynchronized =
            false;

        return false;
    }


    gRemoteClockOffsetUs =
        bestOffset;

    gClockSynchronized =
        true;


    Serial.printf(
        "CLOCK SYNC: "
        "offset=%lld us "
        "RTT=%llu us\n",

        static_cast<
            long long
        >(gRemoteClockOffsetUs),

        static_cast<
            unsigned long long
        >(bestDelay)
    );


    return true;
}


// ============================================================
// LOCAL SERIAL CRC
// ============================================================

static uint16_t crc16(
    const uint8_t *data,
    size_t len)
{
    uint16_t crc = 0xFFFF;


    for (size_t i = 0;
         i < len;
         ++i) {

        crc ^= data[i];


        for (uint8_t j = 0;
             j < 8;
             ++j) {

            if (crc & 1) {

                crc =
                    (crc >> 1) ^
                    0xA001;

            } else {

                crc >>= 1;
            }
        }
    }


    return crc;
}

// LOCAL FRAME SENDER
// ============================================================

static void sendLocalFrame(
    LocalType type,
    const void *payload,
    uint8_t length)
{
    struct LocalHeader {

        uint8_t magic1;

        uint8_t magic2;

        uint8_t type;

        uint8_t length;
    };


    LocalHeader header{
        SERIAL_MAGIC1,
        SERIAL_MAGIC2,
        static_cast<
            uint8_t
        >(type),
        length
    };


    Serial1.write(
        reinterpret_cast<
            uint8_t *
        >(&header),
        sizeof(header)
    );


    if (
        payload &&
        length
    ) {

        Serial1.write(
            static_cast<
                const uint8_t *
            >(payload),
            length
        );
    }


    uint16_t crc =
        crc16(
            reinterpret_cast<
                const uint8_t *
            >(&header),
            sizeof(header)
        );


    if (
        payload &&
        length
    ) {

        const uint8_t *p =
            static_cast<
                const uint8_t *
            >(payload);


        for (uint8_t i = 0;
             i < length;
             ++i) {

            uint8_t b = p[i];

            crc ^= b;


            for (uint8_t j = 0;
                 j < 8;
                 ++j) {

                if (crc & 1) {

                    crc =
                        (crc >> 1) ^
                        0xA001;

                } else {

                    crc >>= 1;
                }
            }
        }
    }


    Serial1.write(
        reinterpret_cast<
            uint8_t *
        >(&crc),
        2
    );


    Serial1.flush();
}

// ============================================================
// BUILD RF TEST PACKET
// ============================================================

static uint16_t buildTestPacket(
    uint8_t *frame,
    uint32_t sequence)
{
    uint16_t p = 0;


    const uint16_t frameControl =
        0x00D0;


    memcpy(
        frame + p,
        &frameControl,
        2
    );

    p += 2;


    uint16_t duration = 0;


    memcpy(
        frame + p,
        &duration,
        2
    );

    p += 2;


    /*
     * IMPORTANT IMPROVEMENT:
     *
     * The destination is the REFERENCE MASTER's MAC, not
     * broadcast.
     *
     * This greatly reduces unrelated receivers being
     * deliberately stimulated.
     *
     * The second reference antenna may also receive it,
     * because it is listening promiscuously.
     */

    memcpy(
        frame + p,
        REFERENCE_MASTER_MAC,
        6
    );

    p += 6;


    // Our source MAC.

    uint8_t ownMac[6];

    esp_wifi_get_mac(
        WIFI_IF_STA,
        ownMac
    );


    memcpy(
        frame + p,
        ownMac,
        6
    );

    p += 6;


    // BSSID = reference master.

    memcpy(
        frame + p,
        REFERENCE_MASTER_MAC,
        6
    );

    p += 6;


    uint16_t seqControl = 0;


    memcpy(
        frame + p,
        &seqControl,
        2
    );

    p += 2;


    // Vendor-specific action.

    frame[p++] = 127;


    // Experimental OUI.

    frame[p++] = 0xDE;

    frame[p++] = 0xAD;

    frame[p++] = 0xBE;


    // Protocol ID.

    frame[p++] = 0x02;


    // "RFTS"

    frame[p++] = 0x52;
    frame[p++] = 0x46;
    frame[p++] = 0x54;
    frame[p++] = 0x53;


    // Run ID.

    memcpy(
        frame + p,
        &gRunId,
        4
    );

    p += 4;


    // Test ID.

    memcpy(
        frame + p,
        &gTestId,
        2
    );

    p += 2;


    // Phase.

    frame[p++] =
        static_cast<
            uint8_t
        >(gPhase);


    // Sequence.

    memcpy(
        frame + p,
        &sequence,
        4
    );

    p += 4;


    // Channel.

    frame[p++] =
        gRFChannel;


    // Fixed deterministic filler.

    while (p < 90) {

        frame[p++] =
            static_cast<
                uint8_t
            >(
                0xA5 ^
                (
                    sequence &
                    0xFF
                )
            );
    }


    return p;
}


// ============================================================
// RF TRANSMISSION
// ============================================================

static void runRFTransmission(
    uint64_t startAtUs)
{
    while (
        static_cast<
            int64_t
        >(
            startAtUs -
            esp_timer_get_time()
        ) > 0
    ) {

        const uint64_t now =
            esp_timer_get_time();

        const uint64_t remaining =
            startAtUs - now;


        if (remaining > 5000) {

            delay(
                static_cast<
                    uint32_t
                >(
                    remaining / 1000
                )
            );

        } else {

            delayMicroseconds(
                static_cast<
                    uint32_t
                >(remaining)
            );
        }
    }


    const uint64_t endAt =
        startAtUs +
        static_cast<uint64_t>(
            gWindowMs
        ) *
        1000ULL;


    uint64_t nextTx =
        startAtUs;


    uint32_t sequence = 0;

    uint8_t frame[128];


    while (
        esp_timer_get_time() <
        endAt
    ) {

        const uint64_t now =
            esp_timer_get_time();


        if (now < nextTx) {

            const uint64_t delayUs =
                nextTx - now;


            if (delayUs > 1000) {

                delay(
                    static_cast<
                        uint32_t
                    >(
                        delayUs /
                        1000
                    )
                );

            } else {

                delayMicroseconds(
                    static_cast<
                        uint32_t
                    >(
                        delayUs
                    )
                );
            }


            continue;
        }


        const uint16_t length =
            buildTestPacket(
                frame,
                sequence
            );


        /*
         * Raw 802.11 action-frame transmission.
         *
         * esp_wifi_80211_tx() is specifically supported for
         * action frames on the ESP32 Wi-Fi driver.
         */

        (void)
            esp_wifi_80211_tx(
                WIFI_IF_STA,
                frame,
                length,
                true
            );


        ++sequence;


        nextTx +=
            gTxIntervalUs;
    }
}

static void runRFReceive(
    uint64_t startAtUs)
{
    while (
        static_cast<
            int64_t
        >(
            startAtUs -
            esp_timer_get_time()
        ) > 0
    ) {

        const uint64_t now =
            esp_timer_get_time();

        const uint64_t remaining =
            startAtUs - now;


        if (remaining > 5000) {

            delay(
                static_cast<
                    uint32_t
                >(
                    remaining / 1000
                )
            );

        } else {

            delayMicroseconds(
                static_cast<
                    uint32_t
                >(remaining)
            );
        }
    }


    const uint64_t endAt =
        startAtUs +
        static_cast<uint64_t>(
            gWindowMs
        ) *
        1000ULL;


    while (
        esp_timer_get_time() <
        endAt
    ) {

        /*
         * No control communication here.
         *
         * The CSI/promiscuous callbacks execute asynchronously.
         */

        delay(1);
    }
}


// ============================================================
// BUILD RESULT
// ============================================================

static RFResultSummary buildResult(
    RadioMode mode)
{
    RFResultSummary r{};


    r.mode =
        static_cast<
            uint8_t
        >(mode);


    portENTER_CRITICAL(
        &gCSIMux
    );


    r.csiFrames =
        static_cast<
            uint16_t
        >(
            min<uint32_t>(
                gCSI.frames,
                UINT16_MAX
            )
        );


    r.csiValid =
        gCSI.frames > 0;


    r.testPackets =
        gPacketStats.packets;


    r.matchingPackets =
        gPacketStats.matchingPackets;


    r.crcPackets =
        gPacketStats.crcPackets;


    r.duplicatePackets =
        gPacketStats.duplicates;


    r.rssiSum =
        gCSI.rssiSum;


    r.rssiSumSq =
        gCSI.rssiSumSq;


    r.rssiMin =
        gCSI.rssiMin;


    r.rssiMax =
        gCSI.rssiMax;


    uint16_t pairs = 0;


    if (gCSI.frames) {

        pairs =
            static_cast<
                uint16_t
            >(
                min<uint32_t>(
                    MAX_CSI_PAIRS,
                    gCSI.pairCountTotal /
                    gCSI.frames
                )
            );
    }


    r.csiPairs =
        pairs;


    for (
        uint8_t out = 0;
        out < EXPORT_CSI_BINS;
        ++out
    ) {


        uint16_t source =
            0;


        if (pairs > 1) {

            source =
                static_cast<
                    uint16_t
                >(
                    (
                        static_cast<
                            uint32_t
                        >(out) *
                        (
                            pairs - 1
                        )
                    ) /
                    (
                        EXPORT_CSI_BINS -
                        1
                    )
                );
        }


        const uint16_t n =
            gCSI.count[source];


        if (n) {

            r.avgI[out] =
                static_cast<
                    int8_t
                >(
                    gCSI.sumI[source] /
                    static_cast<
                        int32_t
                    >(n)
                );


            r.avgQ[out] =
                static_cast<
                    int8_t
                >(
                    gCSI.sumQ[source] /
                    static_cast<
                        int32_t
                    >(n)
                );

        } else {

            r.avgI[out] = 0;

            r.avgQ[out] = 0;
        }
    }


    portEXIT_CRITICAL(
        &gCSIMux
    );


    return r;
}


// ============================================================
// PREPARE
// ============================================================

static bool handlePrepare(
    const PrepareCommand &cmd)
{
    if (
        cmd.magic !=
            RF_MAGIC ||
        cmd.version !=
            PROTOCOL_VERSION ||
        cmd.type !=
            static_cast<
                uint8_t
            >(ControlType::PREPARE)
    ) {

        return false;
    }


    if (
        cmd.channel < 1 ||
        cmd.channel > 13
    ) {

        return false;
    }


    if (
        cmd.windowMs < 50 ||
        cmd.windowMs > 10000
    ) {

        return false;
    }


    gRunId =
        cmd.runId;

    gTestId =
        cmd.testId;

    gPhase =
        static_cast<
            Phase
        >(cmd.phase);

    gRFChannel =
        cmd.channel;

    gWindowMs =
        cmd.windowMs;

    gGuardUs =
        cmd.guardUs;

    gTxIntervalUs =
        cmd.txIntervalUs;

    gTxRate =
        cmd.txRate;

    gBandwidth =
        cmd.bandwidth;


    resetMeasurement();


    /*
     * Prepare slave now, but it must not generate RF traffic.
     */

    prepareLocalSlave();


    /*
     * Configure this radio to RF channel only after all
     * control traffic associated with PREPARE is complete.
     */

    if (!setChannel(
            gRFChannel
        )) {

        return false;
    }


    /*
     * Return to control channel while waiting for START_AT.
     *
     * This is important because START_AT itself arrives via
     * ESP-NOW.
     */

    if (!setChannel(
            CONTROL_CHANNEL
        )) {

        return false;
    }


    return true;
}

// ============================================================
// START_AT
// ============================================================

static bool handleStartAt(
    const StartAtCommand &cmd)
{
    if (
        cmd.magic != RF_MAGIC ||
        cmd.version != PROTOCOL_VERSION ||
        cmd.type != static_cast<
            uint8_t
        >(ControlType::START_AT)
    ) {

        return false;
    }


    if (
        cmd.runId != gRunId ||
        cmd.testId != gTestId
    ) {

        return false;
    }


    if (
        !gClockSynchronized
    ) {

        return false;
    }


    /*
     * Tell the slave its START_AT value first.
     *
     * This MUST happen while there is still plenty of time
     * before startAtUs.
     */

    startLocalSlave(
        cmd.startAtUs
    );


    /*
     * Acknowledge START_AT before changing channel.
     */

    StartAck ack{};

    ack.magic =
        RF_MAGIC;

    ack.version =
        PROTOCOL_VERSION;

    ack.type =
        static_cast<
            uint8_t
        >(ControlType::START_ACK);

    ack.runId =
        gRunId;

    ack.testId =
        gTestId;

    ack.startAtUs =
        cmd.startAtUs;

    ack.status =
        0;


    if (!sendControl(
            &ack,
            sizeof(ack)
        )) {

        return false;
    }


    /*
     * Now there is no further control communication.
     */


    if (!setChannel(
            gRFChannel
        )) {

        return false;
    }


    /*
     * Add a local radio settling period.
     *
     * START_AT supplied by the orchestrator must be
     * sufficiently far in the future to accommodate this.
     */


    const bool transmitting =
        (
            gPhase ==
                Phase::TEST_A_TX ||
            gPhase ==
                Phase::TEST_R_TX
        );


    if (transmitting) {

        runRFTransmission(
            cmd.startAtUs
        );

    } else {

        runRFReceive(
            cmd.startAtUs
        );
    }


    /*
     * Measurement is over.
     *
     * ESP-NOW is still forbidden until we are back on
     * CONTROL_CHANNEL.
     */


    const uint64_t returnTime =
        cmd.startAtUs +
        cmd.windowUs +
        cmd.guardAfterUs;


    while (
        esp_timer_get_time() <
        returnTime
    ) {

        delay(1);
    }


    if (!setChannel(
            CONTROL_CHANNEL
        )) {

        return false;
    }


    delay(30);


    /*
     * Give the UART slave time to send its result.
     */

    const uint32_t waitStart =
        millis();


    while (
        millis() -
        waitStart <
        150
    ) {

        /*
         * UART result parser would run here.
         *
         * Kept isolated from the RF timing path.
         */

        delay(1);
    }


    RFResultSummary localResult =
        buildResult(
            transmitting
            ?
            RadioMode::TX
            :
            RadioMode::RX
        );


    /*
     * Result radio0 = TEST MASTER.
     *
     * radio1 = local test-reference slave.
     *
     * The slave result is inserted by the Serial1 result
     * parser.
     */

    ResultPacket result{};

    result.magic =
        RF_MAGIC;

    result.version =
        PROTOCOL_VERSION;

    result.type =
        static_cast<
            uint8_t
        >(ControlType::RESULT);

    result.runId =
        gRunId;

    result.testId =
        gTestId;

    result.phase =
        static_cast<
            uint8_t
        >(gPhase);

    result.status =
        0;


    result.radio0 =
        localResult;


    /*
     * Until slave result parser is connected:
     *
     * radio1.mode == NONE
     *
     * The proper slave implementation will replace this
     * structure with its measured data.
     */

    result.radio1 = {};


    result.radio1.mode =
        static_cast<
            uint8_t
        >(RadioMode::NONE);


    return sendControl(
        &result,
        sizeof(result)
    );
}


// ============================================================
// PROCESS ESP-NOW
// ============================================================

static void processControl()
{
    EspNowMessage msg{};


    while (
        xQueueReceive(
            gEspNowQueue,
            &msg,
            0
        ) == pdTRUE
    ) {


        if (
            msg.length <
            4
        ) {

            continue;
        }


        const uint8_t type =
            msg.data[3];


        // ----------------------------------------------------
        // SYNC REQUEST
        // ----------------------------------------------------

        if (
            type ==
            static_cast<
                uint8_t
            >(ControlType::SYNC_REQ)
        ) {


            if (
                msg.length <
                sizeof(
                    SyncRequest
                )
            ) {

                continue;
            }


            const SyncRequest *req =
                reinterpret_cast<
                    const SyncRequest *
                >(msg.data);


            const uint64_t t2 =
                esp_timer_get_time();


            SyncResponse resp{};

            resp.magic =
                RF_MAGIC;

            resp.version =
                PROTOCOL_VERSION;

            resp.type =
                static_cast<
                    uint8_t
                >(ControlType::SYNC_RESP);

            resp.sequence =
                req->sequence;

            resp.t1 =
                req->t1;

            resp.t2 =
                t2;


            resp.t3 =
                esp_timer_get_time();


            sendControl(
                &resp,
                sizeof(resp)
            );


            continue;
        }


        // ----------------------------------------------------
        // PREPARE
        // ----------------------------------------------------

        if (
            type ==
            static_cast<
                uint8_t
            >(ControlType::PREPARE)
        ) {


            if (
                msg.length <
                sizeof(
                    PrepareCommand
                )
            ) {

                continue;
            }


            const PrepareCommand *cmd =
                reinterpret_cast<
                    const PrepareCommand *
                >(msg.data);


            if (
                handlePrepare(
                    *cmd
                )
            ) {

                sendReady();
            }


            continue;
        }


        // ----------------------------------------------------
        // START_AT
        // ----------------------------------------------------

        if (
            type ==
            static_cast<
                uint8_t
            >(ControlType::START_AT)
        ) {


            if (
                msg.length <
                sizeof(
                    StartAtCommand
                )
            ) {

                continue;
            }


            const StartAtCommand *cmd =
                reinterpret_cast<
                    const StartAtCommand *
                >(msg.data);


            /*
             * IMPORTANT:
             *
             * handleStartAt does not return until the complete
             * RF experiment is finished and ESP-NOW has returned
             * to the control channel.
             */

            handleStartAt(
                *cmd
            );


            continue;
        }
    }
}

Two further changes are important

The above is the improved test-node master, but there are two corresponding changes I would make to the reference master before calling the protocol complete.

First: START_AT should be expressed in the test master's clock domain, not simply copied from the reference master's esp_timer_get_time(). The reference master should calculate:

\[
T_\text{test}=T_\text{reference}-\Delta t
\]

where Δt is the measured clock offset. The synchronization uses the usual four timestamps:

\[
t_1,t_2,t_3,t_4
\]

and selects the exchange with the lowest round-trip delay. This is the reason for the performClockSync() implementation above.

Second: the slave needs its own START_AT. It receives the same absolute timestamp from its local master through Serial1. Thus:

reference master
      │
      │ START_AT
      ▼
test master
      │
      │ START_AT
      ▼
test-node slave

No UART command is sent during the actual RF window.

Directed test frames

The improved transmitter now addresses the reference master MAC rather than broadcasting:

memcpy(frame + p, REFERENCE_MASTER_MAC, 6);

while the receivers remain in promiscuous mode. ESP-IDF explicitly supports promiscuous packet reception and programmable filtering, and CSI is intended to be used alongside promiscuous mode for this type of measurement. 

I would not attempt to use ESP-NOW's encryption or normal association for the measurement packets. esp_wifi_80211_tx() is intended for raw action frames and non-QoS data frames, and Espressif documents action-frame transmission through this API. 

One correction to the earlier assumption

The ESP32's promiscuous callback can filter the received 802.11 frames, but the CSI callback is a separate data path. Therefore we should not assume that "CSI frame N corresponds exactly to test packet N." The correct measurement model is:

transmitted test packets
                              │
                    ┌─────────┴─────────┐
                    │                   │
              packet callback       CSI callback
                    │                   │
              packet identity        channel data
              sequence number        I/Q samples
              timing/statistics      RSSI
                    │                   │
                    └─────────┬─────────┘
                              │
                         measurement
                           record

That distinction is important for the final data analysis.

The resulting four-channel measurement

For a given frequency/channel, the system will ultimately produce:

RECEIVERS

                    TEST       TEST-REF
                 antenna A     reference
                    │              │
                    ▼              ▼
REF-V TX        CSI/RSSI        CSI/RSSI
REF-H TX        CSI/RSSI        CSI/RSSI


                    RECEIVERS

                  REF-V          REF-H
                    │              │
                    ▼              ▼
TEST-A TX       CSI/RSSI        CSI/RSSI
TEST-REF TX     CSI/RSSI        CSI/RSSI

This is much better than simply comparing RSSI because it gives us a 4×2 directional dataset for every Wi-Fi channel.

The next piece should be the two slave firmwares, because they can then use exactly the same PREPARE → START_AT → fixed RF window → RESULT protocol and make the entire four-ESP32 system deterministic.
