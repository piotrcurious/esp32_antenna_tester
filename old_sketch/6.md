Below is the reference-node slave firmware for the ESP32-WROOM controlling the H-polarised ¼-wave reference antenna.

Its role is deliberately narrow:

REFERENCE NODE

ESP32 MASTER                  ESP32 SLAVE
V antenna                     H antenna
orchestrator                  RF instrument
     │                             │
     └──────── Serial1 ────────────┘

The slave never participates in ESP-NOW. It receives the experiment configuration and timing over Serial1, performs the RF measurement, then returns a compact result.

One important synchronization issue from the previous code is corrected here: an absolute esp_timer_get_time() value belongs to one ESP32's clock domain and cannot simply be copied to another ESP32. Therefore this slave supports a Serial1 clock-synchronization exchange. The master must use that exchange to translate its START_AT timestamp into the slave's clock domain before sending it.

/*
 * ============================================================
 * QUAD ESP32 RF ANTENNA TESTER
 *
 * REFERENCE NODE SLAVE
 *
 * ESP32-WROOM
 *
 * ROLE:
 *   H-polarised 1/4-wave reference antenna
 *
 * CONNECTION:
 *
 *       REFERENCE MASTER
 *          V antenna
 *             |
 *           Serial1
 *             |
 *       REFERENCE SLAVE
 *          H antenna
 *
 * ============================================================
 *
 * This firmware:
 *
 *   - receives experiment commands over Serial1
 *   - never uses ESP-NOW
 *   - performs RF TX or RX depending on phase
 *   - captures CSI
 *   - recognizes the test packet format
 *   - returns a compact RFResultSummary
 *
 * During the actual RF window:
 *
 *   NO Serial1
 *   NO ESP-NOW
 *   NO control traffic
 *
 * ============================================================
 */

// ============================================================
// RESET MEASUREMENTS
// ============================================================

static void resetMeasurements()
{
    portENTER_CRITICAL(
        &gCSIMux
    );

    memset(
        &gCSI,
        0,
        sizeof(gCSI)
    );

    gCSI.rssiMin = 127;

    gCSI.rssiMax = -127;

    memset(
        &gPacketStats,
        0,
        sizeof(gPacketStats)
    );

    portEXIT_CRITICAL(
        &gCSIMux
    );
}


// ============================================================
// CSI CALLBACK
// ============================================================

static void IRAM_ATTR csiCallback(
    void *ctx,
    wifi_csi_info_t *data)
{
    if (data == nullptr) {
        return;
    }

    if (data->buf == nullptr) {
        return;
    }

    if (data->len < 2) {
        return;
    }


    const int pairs =
        min(
            data->len / 2,
            static_cast<int>(
                MAX_CSI_PAIRS
            )
        );


    const int8_t rssi =
        data->rx_ctrl.rssi;


    portENTER_CRITICAL_ISR(
        &gCSIMux
    );


    ++gCSI.frames;


    gCSI.rssiSum +=
        rssi;


    gCSI.rssiSumSq +=
        static_cast<uint32_t>(
            rssi * rssi
        );


    if (rssi <
        gCSI.rssiMin) {

        gCSI.rssiMin =
            rssi;
    }


    if (rssi >
        gCSI.rssiMax) {

        gCSI.rssiMax =
            rssi;
    }


    gCSI.pairCountTotal +=
        pairs;


    for (
        int i = 0;
        i < pairs;
        ++i
    ) {

        /*
         * For the classic ESP32 CSI buffer the data represents
         * signed complex I/Q values.
         *
         * Exact mapping of positions to OFDM subcarriers must
         * be verified against the exact Arduino-ESP32 / IDF
         * version used for the instrument before assigning
         * absolute frequency values.
         */

        const int8_t imag =
            data->buf[
                2 * i
            ];


        const int8_t real =
            data->buf[
                2 * i + 1
            ];


        gCSI.sumI[i] +=
            real;


        gCSI.sumQ[i] +=
            imag;


        ++gCSI.count[i];
    }


    portEXIT_CRITICAL_ISR(
        &gCSIMux
    );
}


// ============================================================
// TEST PACKET FILTER
// ============================================================
//
// Recognizes:
//
//   category = 127
//   OUI      = DE:AD:BE
//   protocol = 02
//   magic    = RFTS
//
// ============================================================

static void promiscuousCallback(
    void *packet,
    wifi_promiscuous_pkt_type_t type)
{
    if (packet == nullptr) {
        return;
    }


    if (
        type != WIFI_PKT_MGMT &&
        type != WIFI_PKT_DATA
    ) {

        return;
    }


    wifi_promiscuous_pkt_t *pkt =
        static_cast<
            wifi_promiscuous_pkt_t *
        >(packet);


    const uint16_t len =
        pkt->rx_ctrl.sig_len;


    if (len < 43) {
        return;
    }


    const uint8_t *frame =
        pkt->payload;


    /*
     * 802.11 frame control.
     */

    const uint16_t fc =
        frame[0] |
        (
            static_cast<
                uint16_t
            >(frame[1])
            << 8
        );


    const uint8_t frameType =
        (fc >> 2) & 0x03;


    const uint8_t subtype =
        (fc >> 4) & 0x0F;


    /*
     * Management Action.
     */

    if (
        frameType != 0 ||
        subtype != 13
    ) {

        return;
    }


    /*
     * Action frame payload:
     *
     * 24 = category
     * 25-27 = OUI
     * 28 = protocol
     * 29-32 = magic
     */

    if (frame[24] != 127) {
        return;
    }


    if (
        frame[25] != 0xDE ||
        frame[26] != 0xAD ||
        frame[27] != 0xBE
    ) {

        return;
    }


    if (frame[28] != 0x02) {
        return;
    }


    if (
        frame[29] != 0x52 ||
        frame[30] != 0x46 ||
        frame[31] != 0x54 ||
        frame[32] != 0x53
    ) {

        return;
    }


    /*
     * Run ID and test ID follow the magic.
     */

    if (len < 42) {
        return;
    }


    uint32_t runId =
        frame[33] |
        (
            static_cast<uint32_t>(
                frame[34]
            ) << 8
        ) |
        (
            static_cast<uint32_t>(
                frame[35]
            ) << 16
        ) |
        (
            static_cast<uint32_t>(
                frame[36]
            ) << 24
        );


    uint16_t testId =
        frame[37] |
        (
            static_cast<uint16_t>(
                frame[38]
            ) << 8
        );


    /*
     * Only accept frames belonging to this experiment.
     */

    if (runId != gRunId ||
        testId != gTestId) {

        return;
    }


    /*
     * Sequence starts at byte 40 in the protocol.
     */

    if (len < 46) {
        return;
    }


    uint32_t sequence =
        frame[40] |
        (
            static_cast<uint32_t>(
                frame[41]
            ) << 8
        ) |
        (
            static_cast<uint32_t>(
                frame[42]
            ) << 16
        ) |
        (
            static_cast<uint32_t>(
                frame[43]
            ) << 24
        );


    portENTER_CRITICAL_ISR(
        &gCSIMux
    );


    ++gPacketStats.packets;


    ++gPacketStats.matchingPackets;


    if (
        gPacketStats.haveSequence &&
        sequence <=
        gPacketStats.lastSequence
    ) {

        ++gPacketStats.duplicates;
    }


    gPacketStats.lastSequence =
        sequence;


    gPacketStats.haveSequence =
        true;


    portEXIT_CRITICAL_ISR(
        &gCSIMux
    );
}


// ============================================================
// WIFI INITIALIZATION
// ============================================================

static void initWiFi()
{
    WiFi.mode(
        WIFI_STA
    );


    WiFi.disconnect(
        true,
        true
    );


    delay(50);


    esp_err_t err;


    err =
        esp_wifi_set_ps(
            WIFI_PS_NONE
        );


    if (err != ESP_OK) {

        Serial.printf(
            "WARNING power save: %s\n",
            esp_err_to_name(err)
        );
    }


    err =
        esp_wifi_set_channel(
            CONTROL_CHANNEL,
            WIFI_SECOND_CHAN_NONE
        );


    if (err != ESP_OK) {

        Serial.printf(
            "FATAL channel: %s\n",
            esp_err_to_name(err)
        );

        while (true) {
            delay(1000);
        }
    }


    err =
        esp_wifi_set_promiscuous(
            true
        );


    if (err != ESP_OK) {

        Serial.printf(
            "FATAL promiscuous: %s\n",
            esp_err_to_name(err)
        );

        while (true) {
            delay(1000);
        }
    }


    err =
        esp_wifi_set_promiscuous_rx_cb(
            promiscuousCallback
        );


    if (err != ESP_OK) {

        Serial.printf(
            "FATAL promisc callback: %s\n",
            esp_err_to_name(err)
        );

        while (true) {
            delay(1000);
        }
    }


    // --------------------------------------------------------
    // CSI
    // --------------------------------------------------------

    err =
        esp_wifi_set_csi_rx_cb(
            csiCallback,
            nullptr
        );


    if (err != ESP_OK) {

        Serial.printf(
            "FATAL CSI callback: %s\n",
            esp_err_to_name(err)
        );

        while (true) {
            delay(1000);
        }
    }


    wifi_csi_config_t config{};


    config.lltf_en =
        true;


    config.htltf_en =
        true;


    config.stbc_htltf2_en =
        true;


    config.ltf_merge_en =
        false;


    config.channel_filter_en =
        false;


    config.manu_scale =
        false;


    config.shift =
        0;


    err =
        esp_wifi_set_csi_config(
            &config
        );


    if (err != ESP_OK) {

        Serial.printf(
            "FATAL CSI config: %s\n",
            esp_err_to_name(err)
        );

        while (true) {
            delay(1000);
        }
    }


    err =
        esp_wifi_set_csi(
            true
        );


    if (err != ESP_OK) {

        Serial.printf(
            "FATAL CSI enable: %s\n",
            esp_err_to_name(err)
        );

        while (true) {
            delay(1000);
        }
    }


    /*
     * This radio can become the H-reference transmitter.
     *
     * TX power is fixed for repeatability.
     */

    err =
        esp_wifi_set_max_tx_power(
            78
        );


    if (err != ESP_OK) {

        Serial.printf(
            "WARNING TX power: %s\n",
            esp_err_to_name(err)
        );
    }
}

static bool setRFChannel(
    uint8_t channel)
{
    const esp_err_t err =
        esp_wifi_set_channel(
            channel,
            WIFI_SECOND_CHAN_NONE
        );


    if (err != ESP_OK) {

        Serial.printf(
            "setChannel(%u) failed: %s\n",
            channel,
            esp_err_to_name(err)
        );

        return false;
    }


    return true;
}


// ============================================================
// CRC16
// ============================================================

static uint16_t crc16(
    const uint8_t *data,
    size_t length)
{
    uint16_t crc =
        0xFFFF;


    for (
        size_t i = 0;
        i < length;
        ++i
    ) {

        crc ^=
            data[i];


        for (
            uint8_t j = 0;
            j < 8;
            ++j
        ) {

            if (crc & 1) {

                crc =
                    (crc >> 1) ^
                    0xA001;

            } else {

                crc >>= 1;
            }
        }
    }


    return crc;
}


// ============================================================
// LOCAL SERIAL PARSER
// ============================================================

class SerialParser {

public:

    enum class State : uint8_t {

        MAGIC1,

        MAGIC2,

        TYPE,

        LENGTH,

        PAYLOAD,

        CRC_LO,

        CRC_HI
    };


    State state =
        State::MAGIC1;


    uint8_t type = 0;

    uint8_t length = 0;

    uint8_t payload[
        SERIAL_MAX_PAYLOAD
    ];

    uint8_t position = 0;

    uint16_t receivedCRC = 0;


    void reset()
    {
        state =
            State::MAGIC1;

        type = 0;

        length = 0;

        position = 0;

        receivedCRC = 0;
    }


    bool feed(
        uint8_t byte)
    {
        switch (state) {


            case State::MAGIC1:

                if (
                    byte ==
                    SERIAL_MAGIC1
                ) {

                    state =
                        State::MAGIC2;
                }

                break;


            case State::MAGIC2:

                if (
                    byte ==
                    SERIAL_MAGIC2
                ) {

                    state =
                        State::TYPE;

                } else {

                    reset();
                }

                break;


            case State::TYPE:

                type =
                    byte;

                state =
                    State::LENGTH;

                break;


            case State::LENGTH:

                length =
                    byte;


                if (
                    length >
                    SERIAL_MAX_PAYLOAD
                ) {

                    reset();

                }
                else if (
                    length == 0
                ) {

                    state =
                        State::CRC_LO;

                }
                else {

                    position = 0;

                    state =
                        State::PAYLOAD;
                }

                break;


            case State::PAYLOAD:

                payload[
                    position++
                ] =
                    byte;


                if (
                    position >=
                    length
                ) {

                    state =
                        State::CRC_LO;
                }

                break;


            case State::CRC_LO:

                receivedCRC =
                    byte;

                state =
                    State::CRC_HI;

                break;


            case State::CRC_HI:

                receivedCRC |=
                    static_cast<
                        uint16_t
                    >(byte) << 8;


                {

                    SerialHeader header{};

                    header.magic1 =
                        SERIAL_MAGIC1;

                    header.magic2 =
                        SERIAL_MAGIC2;

                    header.type =
                        type;

                    header.length =
                        length;


                    uint16_t calculated =
                        crc16(
                            reinterpret_cast<
                                const uint8_t *
                            >(&header),
                            sizeof(header)
                        );


                    for (
                        uint8_t i = 0;
                        i < length;
                        ++i
                    ) {

                        uint8_t b =
                            payload[i];


                        calculated ^=
                            b;


                        for (
                            uint8_t j = 0;
                            j < 8;
                            ++j
                        ) {

                            if (
                                calculated & 1
                            ) {

                                calculated =
                                    (
                                        calculated
                                        >> 1
                                    ) ^
                                    0xA001;

                            } else {

                                calculated >>=
                                    1;
                            }
                        }
                    }


                    const bool good =
                        calculated ==
                        receivedCRC;


                    if (good) {

                        state =
                            State::MAGIC1;

                        return true;
                    }
                }


                reset();

                break;
        }


        return false;
    }
};


static SerialParser gParser;


// ============================================================
// SEND SERIAL FRAME
// ============================================================

static void sendSerialFrame(
    LocalType type,
    const void *payload,
    uint8_t length)
{
    SerialHeader header{};


    header.magic1 =
        SERIAL_MAGIC1;


    header.magic2 =
        SERIAL_MAGIC2;


    header.type =
        static_cast<
            uint8_t
        >(type);


    header.length =
        length;


    Serial1.write(
        reinterpret_cast<
            const uint8_t *
        >(&header),
        sizeof(header)
    );


    if (
        payload &&
        length
    ) {

        Serial1.write(
            static_cast<
                const uint8_t *
            >(payload),
            length
        );
    }


    uint16_t crc =
        crc16(
            reinterpret_cast<
                const uint8_t *
            >(&header),
            sizeof(header)
        );


    if (
        payload &&
        length
    ) {

        const uint8_t *p =
            static_cast<
                const uint8_t *
            >(payload);


        for (
            uint8_t i = 0;
            i < length;
            ++i
        ) {

            uint8_t b =
                p[i];


            crc ^= b;


            for (
                uint8_t j = 0;
                j < 8;
                ++j
            ) {

                if (crc & 1) {

                    crc =
                        (crc >> 1) ^
                        0xA001;

                } else {

                    crc >>= 1;
                }
            }
        }
    }


    Serial1.write(
        reinterpret_cast<
            const uint8_t *
        >(&crc),
        sizeof(crc)
    );


    Serial1.flush();
}


// ============================================================
// PREPARE
// ============================================================

static bool handlePrepare(
    const PrepareCommand &cmd)
{
    if (
        cmd.magic !=
            RF_MAGIC ||
        cmd.version !=
            PROTOCOL_VERSION ||
        cmd.type !=
            static_cast<
                uint8_t
            >(LocalType::PREPARE)
    ) {

        return false;
    }


    if (
        cmd.channel < 1 ||
        cmd.channel > 13
    ) {

        return false;
    }


    if (
        cmd.windowMs < 50 ||
        cmd.windowMs > 10000
    ) {

        return false;
    }


    gRunId =
        cmd.runId;


    gTestId =
        cmd.testId;


    gPhase =
        static_cast<
            Phase
        >(cmd.phase);


    gChannel =
        cmd.channel;


    gWindowMs =
        cmd.windowMs;


    gGuardUs =
        cmd.guardUs;


    gTxIntervalUs =
        cmd.txIntervalUs;


    gBandwidth =
        cmd.bandwidth;


    gTxRate =
        cmd.txRate;


    resetMeasurements();


    /*
     * Do not remain on the RF channel.
     *
     * All configuration communication with the master occurs
     * on Serial1, so this radio can actually be configured
     * now, but returning to the control channel makes the
     * state unambiguous.
     */


    if (
        !setRFChannel(
            CONTROL_CHANNEL
        )
    ) {

        return false;
    }


    gPrepared =
        true;


    return true;
}

        static_cast<
            uint8_t
        >(LocalType::READY);


    ready.runId =
        gRunId;


    ready.testId =
        gTestId;


    ready.status = 0;


    sendSerialFrame(
        LocalType::READY,
        &ready,
        sizeof(ready)
    );
}


// ============================================================
// WAIT UNTIL START
// ============================================================

static void waitUntil(
    uint64_t timestampUs)
{
    while (true) {

        const uint64_t now =
            esp_timer_get_time();


        if (
            now >=
            timestampUs
        ) {

            break;
        }


        const uint64_t remaining =
            timestampUs -
            now;


        if (
            remaining >
            5000
        ) {

            delay(
                static_cast<
                    uint32_t
                >(
                    remaining /
                    1000
                )
            );

        } else {

            delayMicroseconds(
                static_cast<
                    uint32_t
                >(remaining)
            );
        }
    }
}


// ============================================================
// RAW TEST TX PACKET
// ============================================================
//
// Reference-master MAC is carried in the command in the full
// system. For the slave we don't need a separate configurable
// destination if the local master supplies it.
// ============================================================

static uint16_t buildReferenceTxPacket(
    uint8_t *frame,
    uint32_t sequence)
{
    /*
     * The final system should put the actual receiving
     * reference-master MAC into this packet.
     *
     * For now the address fields are broadcast because this
     * slave does not know the master MAC independently.
     *
     * The IMPORTANT filtering identity is the RFTS protocol
     * payload.
     */

    uint16_t p = 0;


    const uint16_t fc =
        0x00D0;


    memcpy(
        frame + p,
        &fc,
        2
    );

    p += 2;


    uint16_t duration = 0;


    memcpy(
        frame + p,
        &duration,
        2
    );

    p += 2;


    memset(
        frame + p,
        0xFF,
        6
    );

    p += 6;


    uint8_t mac[6];


    esp_wifi_get_mac(
        WIFI_IF_STA,
        mac
    );


    memcpy(
        frame + p,
        mac,
        6
    );

    p += 6;


    memset(
        frame + p,
        0xFF,
        6
    );

    p += 6;


    uint16_t seqCtl = 0;


    memcpy(
        frame + p,
        &seqCtl,
        2
    );

    p += 2;


    // Action category

    frame[p++] = 127;


    // Test OUI

    frame[p++] = 0xDE;

    frame[p++] = 0xAD;

    frame[p++] = 0xBE;


    // Protocol revision

    frame[p++] = 0x02;


    // RFTS

    frame[p++] = 0x52;
    frame[p++] = 0x46;
    frame[p++] = 0x54;
    frame[p++] = 0x53;


    // Run ID

    memcpy(
        frame + p,
        &gRunId,
        4
    );

    p += 4;


    // Test ID

    memcpy(
        frame + p,
        &gTestId,
        2
    );

    p += 2;


    // Phase

    frame[p++] =
        static_cast<
            uint8_t
        >(gPhase);


    // Sequence

    memcpy(
        frame + p,
        &sequence,
        4
    );

    p += 4;


    // Channel

    frame[p++] =
        gChannel;


    // H-reference identifier

    frame[p++] = 0x48;


    while (p < 90) {

        frame[p++] =
            static_cast<
                uint8_t
            >(
                0xA5 ^
                (
                    sequence &
                    0xFF
                )
            );
    }


    return p;
}


// ============================================================
// TRANSMIT WINDOW
// ============================================================

static void performTransmit(
    uint64_t startAtUs,
    uint32_t windowUs)
{
    waitUntil(
        startAtUs
    );


    const uint64_t endAtUs =
        startAtUs +
        windowUs;


    uint64_t nextTx =
        startAtUs;


    uint32_t sequence = 0;


    uint8_t frame[128];


    while (
        esp_timer_get_time() <
        endAtUs
    ) {


        const uint64_t now =
            esp_timer_get_time();


        if (
            now <
            nextTx
        ) {

            const uint64_t remaining =
                nextTx -
                now;


            if (
                remaining > 1000
            ) {

                delay(
                    static_cast<
                        uint32_t
                    >(
                        remaining /
                        1000
                    )
                );

            } else {

                delayMicroseconds(
                    static_cast<
                        uint32_t
                    >(
                        remaining
                    )
                );
            }


            continue;
        }


        const uint16_t len =
            buildReferenceTxPacket(
                frame,
                sequence
            );


        esp_wifi_80211_tx(
            WIFI_IF_STA,
            frame,
            len,
            true
        );


        ++sequence;


        nextTx +=
            gTxIntervalUs;
    }
}


// ============================================================
// RECEIVE WINDOW
// ============================================================

static void performReceive(
    uint64_t startAtUs,
    uint32_t windowUs)
{
    waitUntil(
        startAtUs
    );


    const uint64_t endAtUs =
        startAtUs +
        windowUs;


    while (
        esp_timer_get_time() <
        endAtUs
    ) {

        /*
         * CSI and promiscuous packet processing happen through
         * callbacks.
         *
         * Deliberately no Serial1 activity here.
         */

        delay(1);
    }
}


// ============================================================
// BUILD RESULT
// ============================================================

static RFResultSummary buildResult(
    RadioMode mode)
{
    RFResultSummary result{};


    result.mode =
        static_cast<
            uint8_t
        >(mode);


    portENTER_CRITICAL(
        &gCSIMux
    );


    result.csiFrames =
        static_cast<
            uint16_t
        >(
            min<uint32_t>(
                gCSI.frames,
                UINT16_MAX
            )
        );


    result.csiValid =
        gCSI.frames >
        0;


    result.testPackets =
        gPacketStats.packets;


    result.matchingPackets =
        gPacketStats.matchingPackets;


    result.crcPackets =
        gPacketStats.crcPackets;


    result.duplicatePackets =
        gPacketStats.duplicates;


    result.rssiSum =
        gCSI.rssiSum;


    result.rssiSumSq =
        gCSI.rssiSumSq;


    result.rssiMin =
        gCSI.rssiMin;


    result.rssiMax =
        gCSI.rssiMax;


    uint16_t pairs = 0;


    if (
        gCSI.frames
    ) {

        pairs =
            static_cast<
                uint16_t
            >(
                min<uint32_t>(
                    MAX_CSI_PAIRS,
                    gCSI.pairCountTotal /
                    gCSI.frames
                )
            );
    }


    result.csiPairs =
        pairs;


    for (
        uint8_t out = 0;
        out < EXPORT_CSI_BINS;
        ++out
    ) {


        uint16_t source =
            0;


        if (
            pairs > 1
        ) {

            source =
                static_cast<
                    uint16_t
                >(
                    (
                        static_cast<
                            uint32_t
                        >(out) *
                        (
                            pairs - 1
                        )
                    ) /
                    (
                        EXPORT_CSI_BINS -
                        1
                    )
                );
        }


        const uint16_t n =
            gCSI.count[
                source
            ];


        if (n) {

            result.avgI[out] =
                static_cast<
                    int8_t
                >(
                    gCSI.sumI[
                        source
                    ] /
                    static_cast<
                        int32_t
                    >(n)
                );


            result.avgQ[out] =
                static_cast<
                    int8_t
                >(
                    gCSI.sumQ[
                        source
                    ] /
                    static_cast<
                        int32_t
                    >(n)
                );

        } else {

            result.avgI[out] = 0;

            result.avgQ[out] = 0;
        }
    }


    portEXIT_CRITICAL(
        &gCSIMux
    );


    return result;
}


// ============================================================
// SEND RESULT
// ============================================================

static void sendResult()
{
    const bool txPhase =
        (
            gPhase ==
                Phase::REF_H_TX
        );


    RFResultSummary result =
        buildResult(
            txPhase
            ?
            RadioMode::TX
            :
            RadioMode::RX
        );


    sendSerialFrame(
        LocalType::RESULT,
        &result,
        sizeof(result)
    );
}


// ============================================================
// START_AT
// ============================================================

static bool handleStartAt(
    const StartAtCommand &cmd)
{
    if (
        !gPrepared
    ) {

        return false;
    }


    if (
        cmd.magic !=
            RF_MAGIC ||
        cmd.version !=
            PROTOCOL_VERSION ||
        cmd.type !=
            static_cast<
                uint8_t
            >(LocalType::START_AT)
    ) {

        return false;
    }


    if (
        cmd.runId !=
            gRunId ||
        cmd.testId !=
            gTestId
    ) {

        return false;
    }


    /*
     * Configure the RF channel BEFORE the absolute start
     * timestamp.
     */

    if (
        !setRFChannel(
            gChannel
        )
    ) {

        return false;
    }


    gRunning =
        true;


    resetMeasurements();


    /*
     * Reference H antenna transmits only during:
     *
     *      REF_H_TX
     *
     * Everything else means the H reference antenna is an
     * RX instrument.
     */

    const bool transmitting =
        (
            gPhase ==
                Phase::REF_H_TX
        );


    if (
        transmitting
    ) {

        performTransmit(
            cmd.startAtUs,
            cmd.windowUs
        );

    } else {

        performReceive(
            cmd.startAtUs,
            cmd.windowUs
        );
    }


    /*
     * End-of-window guard.
     *
     * Still no Serial1 traffic until the RF window is
     * definitively over.
     */

    const uint64_t returnAt =
        cmd.startAtUs +
        cmd.windowUs +
        cmd.guardAfterUs;


    while (
        esp_timer_get_time() <
        returnAt
    ) {

        delay(1);
    }


    /*
     * Return to control/default channel.
     */

    if (
        !setRFChannel(
            CONTROL_CHANNEL
        )
    ) {

        gRunning =
            false;

        return false;
    }


    gRunning =
        false;


    /*
     * Only NOW is Serial1 traffic allowed again.
     */

    delay(5);


    sendResult();


    gPrepared =
        false;


    return true;
}


// ============================================================
// CLOCK SYNCHRONIZATION
// ============================================================
//
// The local master can send:
//   SYNC_REQ(t1)
//
// This slave returns:
//   t1
//   t2 = receive time
//   t3 = transmit time
//
// The master computes:
//   offset = ((t2-t1)+(t3-t4))/2
//
// and therefore can convert:
//
//   master_time -> slave_time
//
// ============================================================

static void handleSyncRequest(
    const SyncRequest &req)
{
    if (
        req.magic !=
            RF_MAGIC ||
        req.version !=
            PROTOCOL_VERSION ||
        req.type !=
            static_cast<
                uint8_t
            >(LocalType::SYNC_REQ)
    ) {

        return;
    }


    const uint64_t t2 =
        esp_timer_get_time();


    SyncResponse response{};


    response.magic =
        RF_MAGIC;


    response.version =
        PROTOCOL_VERSION;


    response.type =
        static_cast<
            uint8_t
        >(LocalType::SYNC_RESP);


    response.sequence =
        req.sequence;


    response.t1 =
        req.t1;


    response.t2 =
        t2;


    /*
     * Timestamp immediately before UART transmission.
     */

    response.t3 =
        esp_timer_get_time();


    sendSerialFrame(
        LocalType::SYNC_RESP,
        &response,
        sizeof(response)
    );
}


// ============================================================
// ABORT
// ============================================================

static void handleAbort()
{
    gRunning =
        false;

    gPrepared =
        false;


    resetMeasurements();


    setRFChannel(
        CONTROL_CHANNEL
    );
}


// ============================================================
// PROCESS COMPLETE SERIAL FRAME
// ============================================================

static void processSerialFrame()
{
    switch (
        static_cast<
            LocalType
        >(gParser.type)
    ) {


        // ----------------------------------------------------
        // PREPARE
        // ----------------------------------------------------

        case LocalType::PREPARE:

            if (
                gParser.length >=
                sizeof(
                    PrepareCommand
                )
            ) {

                PrepareCommand cmd{};


                memcpy(
                    &cmd,
                    gParser.payload,
                    sizeof(cmd)
                );


                if (
                    handlePrepare(
                        cmd
                    )
                ) {

                    sendReady();
                }
            }

            break;


        // ----------------------------------------------------
        // START_AT
        // ----------------------------------------------------

        case LocalType::START_AT:

            if (
                gParser.length >=
                sizeof(
                    StartAtCommand
                )
            ) {

                StartAtCommand cmd{};


                memcpy(
                    &cmd,
                    gParser.payload,
                    sizeof(cmd)
                );


                handleStartAt(
                    cmd
                );
            }

            break;


        // ----------------------------------------------------
        // SYNC
        // ----------------------------------------------------

        case LocalType::SYNC_REQ:

            if (
                gParser.length >=
                sizeof(
                    SyncRequest
                )
            ) {

                SyncRequest req{};


                memcpy(
                    &req,
                    gParser.payload,
                    sizeof(req)
                );


                handleSyncRequest(
                    req
                );
            }

            break;


        // ----------------------------------------------------
        // ABORT
        // ----------------------------------------------------

        case LocalType::ABORT:

            handleAbort();

            break;


        default:

            break;
    }
}


// ============================================================
// UART POLLER
// ============================================================

static void pollSerial()
{
    /*
     * During the RF window no Serial1 bytes are expected.
     */

    if (gRunning) {
        return;
    }


    while (
        Serial1.available()
    ) {

        const int value =
            Serial1.read();


        if (
            value < 0
        ) {

            break;
        }


        if (
            gParser.feed(
                static_cast<
                    uint8_t
                >(value)
            )
        ) {

            processSerialFrame();

            gParser.reset();
        }
    }
}

// SETUP

void setup()
{
    Serial.begin(
        115200
    );


    delay(1000);


    Serial.println();

    Serial.println(
        "========================================"
    );

    Serial.println(
        "QUAD ESP32 RF TESTER"
    );

    Serial.println(
        "REFERENCE NODE SLAVE"
    );

    Serial.println(
        "ESP32-WROOM"
    );

    Serial.println(
        "H-POL 1/4-WAVE REFERENCE ANTENNA"
    );

    Serial.println(
        "========================================"
    );


    // --------------------------------------------------------
    // Serial1 to master
    // --------------------------------------------------------

    Serial1.begin(
        SERIAL_BAUD,
        SERIAL_8N1,
        SERIAL_RX_PIN,
        SERIAL_TX_PIN
    );


    delay(100);


    // --------------------------------------------------------
    // Wi-Fi
    // --------------------------------------------------------

    initWiFi();


    // --------------------------------------------------------
    // MAC
    // --------------------------------------------------------

    uint8_t mac[6];


    esp_wifi_get_mac(
        WIFI_IF_STA,
        mac
    );


    Serial.printf(
        "SLAVE MAC: "
        "%02X:%02X:%02X:%02X:%02X:%02X\n",

        mac[0],
        mac[1],
        mac[2],
        mac[3],
        mac[4],
        mac[5]
    );


    Serial.println(
        "READY"
    );
}


// ============================================================
// LOOP
// ============================================================

void loop()
{
    if (!gRunning) {

        pollSerial();
    }


    /*
     * Do NOT use long delay() here.
     *
     * The serial parser should remain responsive while waiting
     * for synchronization and experiment commands.
     */

    delay(1);
}

Resulting role assignment

With this slave installed, the four radios are now:

REFERENCE NODE
────────────────────────────────

Master
  antenna = 1/4 λ vertical
  role    = orchestrator + RF instrument

Slave
  antenna = 1/4 λ horizontal
  role    = RF instrument


TEST NODE
────────────────────────────────

Master
  antenna = test antenna
  role    = RF instrument

Slave
  antenna = reference antenna
  role    = RF instrument

And the four experiments become:

Phase 0: REF_V_TX

REF-V ───────────────► TEST-A
                     ► TEST-REF


Phase 1: REF_H_TX

REF-H ───────────────► TEST-A
                     ► TEST-REF


Phase 2: TEST_A_TX

TEST-A ──────────────► REF-V
                     ► REF-H


Phase 3: TEST_R_TX

TEST-REF ────────────► REF-V
                     ► REF-H

That gives the useful matrix:

\[
\begin{bmatrix}
H_{\mathrm{V\rightarrow A}} & H_{\mathrm{V\rightarrow R}}\\
H_{\mathrm{H\rightarrow A}} & H_{\mathrm{H\rightarrow R}}
\end{bmatrix}
\]

and, from the reverse measurements,

\[
\begin{bmatrix}
H_{\mathrm{A\rightarrow V}} & H_{\mathrm{A\rightarrow H}}\\
H_{\mathrm{R\rightarrow V}} & H_{\mathrm{R\rightarrow H}}
\end{bmatrix}.
\]

The next important piece is the test-node slave, after which all four firmware components will share one protocol. At that point I would also make one final protocol correction: have the masters exchange the actual MAC addresses of all four radios, so the test frames can be truly directed rather than the reference slave's temporary broadcast frame.
