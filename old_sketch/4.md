Below is the ESP32-WROOM test-node master. It mirrors the reference master protocol: its own RF chain is the test antenna, its Serial1 slave controls the test-node reference antenna, and it accepts PREPARE/GO/ABORT over ESP-NOW.

One important design point: the test-node master does not initiate any ESP-NOW traffic during the RF window. It switches to the requested RF channel, performs TX or CSI RX for the fixed duration, returns to the control channel, and only then sends its result.

/*
 * ============================================================
 * QUAD ESP32 RF ANTENNA TESTER
 *
 * TEST NODE MASTER
 * ESP32-WROOM
 * ============================================================
 *
 * Hardware:
 *
 *             TEST NODE
 *
 *        ESP32 MASTER
 *        TEST ANTENNA
 *             |
 *           Serial1
 *             |
 *        ESP32 SLAVE
 *        REFERENCE ANTENNA
 *
 *
 * Remote:
 *
 *        REFERENCE NODE MASTER
 *        V POLARISATION
 *
 *
 * The REFERENCE NODE is the experiment orchestrator.
 *
 * This node:
 *
 *   - accepts PREPARE over ESP-NOW
 *   - configures its local RF chain
 *   - configures its Serial1 slave
 *   - acknowledges GO
 *   - switches to the RF test channel
 *   - performs TX or CSI RX
 *   - returns to control channel
 *   - sends RF results to reference master
 *
 * IMPORTANT:
 *
 * ESP-NOW is NEVER used during RF measurement.
 *
 * ============================================================
 */

// ============================================================
// CSI ACCUMULATOR
// ============================================================

struct CsiAccumulator {

    volatile uint32_t frames;

    volatile int32_t rssiSum;

    volatile uint32_t rssiSumSq;

    volatile int8_t rssiMin;

    volatile int8_t rssiMax;

    volatile uint32_t pairCountTotal;

    int32_t sumI[MAX_CSI_PAIRS];

    int32_t sumQ[MAX_CSI_PAIRS];

    uint16_t binCount[MAX_CSI_PAIRS];
};


static CsiAccumulator gCsi;

static portMUX_TYPE gCsiMux =
    portMUX_INITIALIZER_UNLOCKED;


// ============================================================
// RESET CSI
// ============================================================

static void resetCsi()
{
    portENTER_CRITICAL(&gCsiMux);

    memset(
        &gCsi,
        0,
        sizeof(gCsi)
    );

    gCsi.rssiMin = 127;

    gCsi.rssiMax = -127;

    portEXIT_CRITICAL(&gCsiMux);
}


// ============================================================
// CSI CALLBACK
// ============================================================
//
// Keep this extremely short.
//
// NO:
//   Serial
//   malloc
//   String
//   vectors
//   floating-point calculations
//
// ============================================================

static void IRAM_ATTR onCsiData(
    void *ctx,
    wifi_csi_info_t *data)
{
    if (data == nullptr) {
        return;
    }

    if (data->buf == nullptr) {
        return;
    }

    if (data->len < 2) {
        return;
    }

    const int pairCount =
        min(
            data->len / 2,
            static_cast<int>(
                MAX_CSI_PAIRS
            )
        );

    const int8_t rssi =
        data->rx_ctrl.rssi;

    portENTER_CRITICAL_ISR(
        &gCsiMux
    );

    gCsi.frames++;

    gCsi.rssiSum += rssi;

    gCsi.rssiSumSq +=
        static_cast<uint32_t>(
            rssi * rssi
        );

    if (rssi < gCsi.rssiMin) {
        gCsi.rssiMin = rssi;
    }

    if (rssi > gCsi.rssiMax) {
        gCsi.rssiMax = rssi;
    }

    gCsi.pairCountTotal +=
        pairCount;

    for (int i = 0;
         i < pairCount;
         ++i) {

        const int8_t imag =
            data->buf[2 * i];

        const int8_t real =
            data->buf[2 * i + 1];

        gCsi.sumI[i] += real;

        gCsi.sumQ[i] += imag;

        gCsi.binCount[i]++;
    }

    portEXIT_CRITICAL_ISR(
        &gCsiMux
    );
}


// ============================================================
// WIFI SETUP
// ============================================================

static void initWifi()
{
    WiFi.mode(WIFI_STA);

    WiFi.disconnect(
        true,
        true
    );

    delay(50);

    esp_err_t err;

    err = esp_wifi_set_ps(
        WIFI_PS_NONE
    );

    if (err != ESP_OK) {

        Serial.printf(
            "WARNING power-save: %s\n",
            esp_err_to_name(err)
        );
    }


    err = esp_wifi_set_channel(
        CONTROL_CHANNEL,
        WIFI_SECOND_CHAN_NONE
    );

    if (err != ESP_OK) {

        Serial.printf(
            "FATAL set channel: %s\n",
            esp_err_to_name(err)
        );

        while (true) {
            delay(1000);
        }
    }


    err =
        esp_wifi_set_promiscuous(
            true
        );

    if (err != ESP_OK) {

        Serial.printf(
            "FATAL promiscuous: %s\n",
            esp_err_to_name(err)
        );

        while (true) {
            delay(1000);
        }
    }


    // --------------------------------------------------------
    // CSI callback
    // --------------------------------------------------------

    err =
        esp_wifi_set_csi_rx_cb(
            onCsiData,
            nullptr
        );

    if (err != ESP_OK) {

        Serial.printf(
            "FATAL CSI callback: %s\n",
            esp_err_to_name(err)
        );

        while (true) {
            delay(1000);
        }
    }


    wifi_csi_config_t config{};

    config.lltf_en = true;

    config.htltf_en = true;

    config.stbc_htltf2_en = true;

    config.ltf_merge_en = false;

    config.channel_filter_en = false;

    config.manu_scale = false;

    config.shift = 0;


    err =
        esp_wifi_set_csi_config(
            &config
        );

    if (err != ESP_OK) {

        Serial.printf(
            "FATAL CSI config: %s\n",
            esp_err_to_name(err)
        );

        while (true) {
            delay(1000);
        }
    }


    err =
        esp_wifi_set_csi(true);

    if (err != ESP_OK) {

        Serial.printf(
            "FATAL CSI enable: %s\n",
            esp_err_to_name(err)
        );

        while (true) {
            delay(1000);
        }
    }


    // Conservative initial value.
    //
    // 78 = 19.5 dBm in the ESP-IDF quarter-dBm
    // representation, subject to the particular ESP32
    // hardware's actual limits.
    //
    // Final EIRP must account for antenna gain.

    err =
        esp_wifi_set_max_tx_power(
            78
        );

    if (err != ESP_OK) {

        Serial.printf(
            "WARNING TX power: %s\n",
            esp_err_to_name(err)
        );
    }
}


// ============================================================
// ESP-NOW RX CALLBACK
// ============================================================

static void onEspNowReceive(
    const esp_now_recv_info_t *info,
    const uint8_t *data,
    int len)
{
    if (info == nullptr ||
        data == nullptr) {
        return;
    }

    if (memcmp(
            info->src_addr,
            REFERENCE_MASTER_MAC,
            6
        ) != 0) {

        return;
    }


    if (len <= 0 ||
        len >
        static_cast<int>(
            sizeof(
                EspNowRxMessage::data
            )
        )) {

        return;
    }


    EspNowRxMessage msg{};

    memcpy(
        msg.srcMac,
        info->src_addr,
        6
    );

    msg.length =
        static_cast<uint16_t>(
            len
        );

    memcpy(
        msg.data,
        data,
        len
    );


    if (gEspNowQueue != nullptr) {

        xQueueSend(
            gEspNowQueue,
            &msg,
            0
        );
    }
}


// ============================================================
// ADD ESP-NOW PEER
// ============================================================

static void initEspNow()
{
    gEspNowQueue =
        xQueueCreate(
            8,
            sizeof(
                EspNowRxMessage
            )
        );


    if (gEspNowQueue == nullptr) {

        Serial.println(
            "FATAL: ESP-NOW queue"
        );

        while (true) {
            delay(1000);
        }
    }


    esp_err_t err =
        esp_now_init();

    if (err != ESP_OK) {

        Serial.printf(
            "FATAL esp_now_init: %s\n",
            esp_err_to_name(err)
        );

        while (true) {
            delay(1000);
        }
    }


    esp_now_set_pmk(
        ESPNOW_PMK
    );


    esp_now_register_recv_cb(
        onEspNowReceive
    );


    if (!esp_now_is_peer_exist(
            REFERENCE_MASTER_MAC
        )) {

        esp_now_peer_info_t peer{};

        memcpy(
            peer.peer_addr,
            REFERENCE_MASTER_MAC,
            6
        );

        peer.channel =
            CONTROL_CHANNEL;

        peer.ifidx =
            WIFI_IF_STA;

        peer.encrypt = false;


        err =
            esp_now_add_peer(
                &peer
            );

        if (err != ESP_OK) {

            Serial.printf(
                "FATAL add peer: %s\n",
                esp_err_to_name(err)
            );

            while (true) {
                delay(1000);
            }
        }
    }
}

// ============================================================
// ESP-NOW SEND
// ============================================================

static bool sendControl(
    const void *data,
    size_t len)
{
    if (gChannel !=
        CONTROL_CHANNEL) {

        Serial.println(
            "ERROR: ESP-NOW outside control channel"
        );

        return false;
    }


    const esp_err_t err =
        esp_now_send(
            REFERENCE_MASTER_MAC,
            static_cast<
                const uint8_t *
            >(data),
            len
        );


    if (err != ESP_OK) {

        Serial.printf(
            "esp_now_send: %s\n",
            esp_err_to_name(err)
        );

        return false;
    }

    return true;
}

// ============================================================
// LOCAL SERIAL CRC
// ============================================================

static uint16_t crc16(
    const uint8_t *data,
    size_t len)
{
    uint16_t crc = 0xFFFF;

    for (size_t i = 0;
         i < len;
         ++i) {

        crc ^= data[i];

        for (uint8_t j = 0;
             j < 8;
             ++j) {

            if (crc & 1) {

                crc =
                    (crc >> 1) ^
                    0xA001;

            } else {

                crc >>= 1;
            }
        }
    }

    return crc;
}


// ============================================================
// LOCAL SERIAL FRAME
// ============================================================

static void sendLocalFrame(
    LocalCmdType type,
    const uint8_t *payload,
    uint8_t length)
{
    if (length >
        LOCAL_MAX_PAYLOAD) {

        return;
    }


    LocalFrameHeader header{};

    header.magic1 =
        LOCAL_MAGIC1;

    header.magic2 =
        LOCAL_MAGIC2;

    header.type =
        static_cast<uint8_t>(
            type
        );

    header.length =
        length;


    Serial1.write(
        reinterpret_cast<
            uint8_t *
        >(&header),
        sizeof(header)
    );


    if (length > 0 &&
        payload != nullptr) {

        Serial1.write(
            payload,
            length
        );
    }


    uint16_t crc =
        crc16(
            reinterpret_cast<
                const uint8_t *
            >(&header),
            sizeof(header)
        );


    if (length > 0 &&
        payload != nullptr) {

        for (uint8_t i = 0;
             i < length;
             ++i) {

            const uint8_t b =
                payload[i];

            crc ^= b;

            for (uint8_t j = 0;
                 j < 8;
                 ++j) {

                if (crc & 1) {

                    crc =
                        (crc >> 1) ^
                        0xA001;

                } else {

                    crc >>= 1;
                }
            }
        }
    }


    Serial1.write(
        reinterpret_cast<
            uint8_t *
        >(&crc),
        sizeof(crc)
    );

    Serial1.flush();
}

// ============================================================
// BUILD RAW TEST PACKET
// ============================================================

static uint16_t buildTestFrame(
    uint8_t *frame,
    uint32_t sequence)
{
    uint16_t p = 0;


    // --------------------------------------------------------
    // 802.11 management Action frame
    // --------------------------------------------------------

    const uint16_t frameControl =
        0x00D0;


    memcpy(
        frame + p,
        &frameControl,
        2
    );

    p += 2;


    uint16_t duration = 0;


    memcpy(
        frame + p,
        &duration,
        2
    );

    p += 2;


    // --------------------------------------------------------
    // Destination = broadcast
    // --------------------------------------------------------

    memset(
        frame + p,
        0xFF,
        6
    );

    p += 6;


    // --------------------------------------------------------
    // Source
    // --------------------------------------------------------

    uint8_t mac[6];

    esp_wifi_get_mac(
        WIFI_IF_STA,
        mac
    );


    memcpy(
        frame + p,
        mac,
        6
    );

    p += 6;


    // --------------------------------------------------------
    // BSSID
    // --------------------------------------------------------

    memset(
        frame + p,
        0xFF,
        6
    );

    p += 6;


    // Sequence control.
    //
    // Driver will normally manage this field.

    uint16_t seqControl = 0;


    memcpy(
        frame + p,
        &seqControl,
        2
    );

    p += 2;


    // --------------------------------------------------------
    // Vendor-specific action
    // --------------------------------------------------------

    frame[p++] = 127;


    // Test OUI

    frame[p++] = 0xDE;

    frame[p++] = 0xAD;

    frame[p++] = 0xBE;


    // Test protocol version

    frame[p++] = 0x01;


    // Magic "RFTS"

    frame[p++] = 0x52;

    frame[p++] = 0x46;

    frame[p++] = 0x54;

    frame[p++] = 0x53;


    // Run ID

    memcpy(
        frame + p,
        &gRunId,
        sizeof(gRunId)
    );

    p += sizeof(gRunId);


    // Test ID

    memcpy(
        frame + p,
        &gTestId,
        sizeof(gTestId)
    );

    p += sizeof(gTestId);


    // Phase

    frame[p++] =
        static_cast<uint8_t>(
            gPhase
        );


    // Sequence

    memcpy(
        frame + p,
        &sequence,
        sizeof(sequence)
    );

    p += sizeof(sequence);


    // Deterministic filler

    while (p < 64) {

        frame[p++] =
            0xA5;
    }


    return p;
}


// ============================================================
// RAW RF TRANSMISSION
// ============================================================

static bool transmitRF()
{
    uint8_t frame[96];


    const uint64_t startUs =
        esp_timer_get_time();


    const uint64_t endUs =
        startUs +
        static_cast<uint64_t>(
            gDurationMs
        ) *
        1000ULL;


    uint64_t nextTx =
        startUs;


    uint32_t sequence = 0;

    uint32_t sent = 0;

    uint32_t errors = 0;


    while (
        esp_timer_get_time() <
        endUs) {


        const uint64_t now =
            esp_timer_get_time();


        if (now < nextTx) {

            const uint64_t waitUs =
                nextTx - now;


            if (waitUs > 1000) {

                delay(
                    static_cast<
                        uint32_t
                    >(waitUs / 1000)
                );

            } else {

                delayMicroseconds(
                    static_cast<
                        uint32_t
                    >(waitUs)
                );
            }


            continue;
        }


        const uint16_t len =
            buildTestFrame(
                frame,
                sequence
            );


        const esp_err_t err =
            esp_wifi_80211_tx(
                WIFI_IF_STA,
                frame,
                len,
                true
            );


        if (err == ESP_OK) {

            ++sent;

        } else {

            ++errors;
        }


        ++sequence;


        nextTx +=
            gTxIntervalUs;
    }


    Serial.printf(
        "RF TX: packets=%lu "
        "errors=%lu\n",

        static_cast<
            unsigned long
        >(sent),

        static_cast<
            unsigned long
        >(errors)
    );


    return true;
}


// ============================================================
// CSI RESULT
// ============================================================

static RFResultSummary buildResult(
    RadioMode mode)
{
    RFResultSummary result{};

    result.mode =
        static_cast<uint8_t>(
            mode
        );


    portENTER_CRITICAL(
        &gCsiMux
    );


    result.csiFrames =
        static_cast<uint16_t>(
            min<uint32_t>(
                gCsi.frames,
                UINT16_MAX
            )
        );


    result.csiValid =
        gCsi.frames >
        0;


    result.rssiSum =
        gCsi.rssiSum;


    result.rssiSumSq =
        gCsi.rssiSumSq;


    result.rssiMin =
        gCsi.rssiMin;


    result.rssiMax =
        gCsi.rssiMax;


    uint16_t averagePairs = 0;


    if (gCsi.frames > 0) {

        averagePairs =
            static_cast<uint16_t>(
                min<uint32_t>(
                    MAX_CSI_PAIRS,
                    gCsi.pairCountTotal /
                    gCsi.frames
                )
            );
    }


    result.csiPairs =
        averagePairs;


    for (uint8_t out = 0;
         out < EXPORT_CSI_BINS;
         ++out) {


        uint16_t sourceIndex = 0;


        if (averagePairs > 1) {

            sourceIndex =
                static_cast<uint16_t>(
                    (
                        static_cast<
                            uint32_t
                        >(out) *
                        (
                            averagePairs -
                            1
                        )
                    ) /
                    (
                        EXPORT_CSI_BINS -
                        1
                    )
                );
        }


        const uint16_t count =
            gCsi.binCount[
                sourceIndex
            ];


        if (count > 0) {

            result.avgI[out] =
                static_cast<int8_t>(
                    gCsi.sumI[
                        sourceIndex
                    ] /
                    count
                );


            result.avgQ[out] =
                static_cast<int8_t>(
                    gCsi.sumQ[
                        sourceIndex
                    ] /
                    count
                );

        } else {

            result.avgI[out] = 0;

            result.avgQ[out] = 0;
        }
    }


    portEXIT_CRITICAL(
        &gCsiMux
    );


    return result;
}


// ============================================================
// SEND RESULT
// ============================================================

static bool sendResult(
    const RFResultSummary &localResult)
{
    /*
     * The remote node has two radios.
     *
     * radio0 = its MASTER / TEST ANTENNA
     *
     * radio1 = its SERIAL1 SLAVE / REFERENCE ANTENNA
     *
     * During this first implementation the slave result is
     * inserted by the local UART result handler.
     */

    RemoteResultPacket result{};

    result.magic = 0x5246;

    result.version = 1;

    result.type =
        static_cast<uint8_t>(
            RemoteMsgType::RESULT
        );

    result.runId =
        gRunId;

    result.testId =
        gTestId;

    result.phase =
        static_cast<uint8_t>(
            gPhase
        );

    result.status = 0;


    // Radio 0 = this master.

    result.radio0 =
        localResult;


    // Radio 1 = local Serial1 slave.
    //
    // This will be populated below when the slave-result
    // parser is implemented.

    memset(
        &result.radio1,
        0,
        sizeof(result.radio1)
    );


    result.radio1.mode =
        static_cast<uint8_t>(
            RadioMode::NONE
        );


    return sendControl(
        &result,
        sizeof(result)
    );
}


// ============================================================
// LOCAL SLAVE RESULT PARSER
// ============================================================
//
// This is deliberately implemented as a byte-stream parser.
// Serial1 can be interrupted or fragmented arbitrarily.
//
// ============================================================

class LocalSerialParser {

public:

    enum class State : uint8_t {

        WAIT_MAGIC1,

        WAIT_MAGIC2,

        TYPE,

        LENGTH,

        PAYLOAD,

        CRC_LO,

        CRC_HI
    };


    State state =
        State::WAIT_MAGIC1;


    uint8_t type = 0;

    uint8_t length = 0;

    uint8_t payload[
        LOCAL_MAX_PAYLOAD
    ];

    uint8_t position = 0;

    uint16_t receivedCrc = 0;


    void reset()
    {
        state =
            State::WAIT_MAGIC1;

        type = 0;

        length = 0;

        position = 0;

        receivedCrc = 0;
    }


    bool feed(
        uint8_t byte)
    {
        switch (state) {


            case State::WAIT_MAGIC1:

                if (byte ==
                    LOCAL_MAGIC1) {

                    state =
                        State::WAIT_MAGIC2;
                }

                break;


            case State::WAIT_MAGIC2:

                if (byte ==
                    LOCAL_MAGIC2) {

                    state =
                        State::TYPE;

                } else {

                    reset();
                }

                break;


            case State::TYPE:

                type = byte;

                state =
                    State::LENGTH;

                break;


            case State::LENGTH:

                length = byte;

                if (length >
                    LOCAL_MAX_PAYLOAD) {

                    reset();

                } else if (
                    length == 0) {

                    state =
                        State::CRC_LO;

                } else {

                    position = 0;

                    state =
                        State::PAYLOAD;
                }

                break;


            case State::PAYLOAD:

                payload[position++] =
                    byte;

                if (position >= length) {

                    state =
                        State::CRC_LO;
                }

                break;


            case State::CRC_LO:

                receivedCrc =
                    byte;

                state =
                    State::CRC_HI;

                break;


            case State::CRC_HI:

                receivedCrc |=
                    static_cast<
                        uint16_t
                    >(byte) << 8;

                {

                    LocalFrameHeader header{};

                    header.magic1 =
                        LOCAL_MAGIC1;

                    header.magic2 =
                        LOCAL_MAGIC2;

                    header.type =
                        type;

                    header.length =
                        length;


                    uint16_t crc =
                        crc16(
                            reinterpret_cast<
                                const uint8_t *
                            >(&header),
                            sizeof(header)
                        );


                    for (uint8_t i = 0;
                         i < length;
                         ++i) {

                        uint8_t b =
                            payload[i];

                        crc ^= b;

                        for (uint8_t j = 0;
                             j < 8;
                             ++j) {

                            if (crc & 1) {

                                crc =
                                    (crc >> 1) ^
                                    0xA001;

                            } else {

                                crc >>= 1;
                            }
                        }
                    }


                    const bool valid =
                        crc ==
                        receivedCrc;


                    if (valid) {

                        state =
                            State::WAIT_MAGIC1;

                        return true;
                    }
                }


                reset();

                break;
        }


        return false;
    }
};


static LocalSerialParser gLocalParser;


// ============================================================
// LOCAL RESULT STORAGE
// ============================================================

static RFResultSummary gLocalSlaveResult{};

static bool gLocalSlaveResultValid =
    false;


// ============================================================
// POLL LOCAL SERIAL
// ============================================================

static void pollLocalSerial()
{
    while (Serial1.available()) {

        const int value =
            Serial1.read();

        if (value < 0) {
            break;
        }


        if (gLocalParser.feed(
                static_cast<uint8_t>(
                    value
                )
            )) {


            if (
                gLocalParser.type ==
                static_cast<uint8_t>(
                    LocalMsgType::RESULT
                )
            ) {


                if (
                    gLocalParser.length >=
                    sizeof(
                        RFResultSummary
                    )
                ) {

                    memcpy(
                        &gLocalSlaveResult,
                        gLocalParser.payload,
                        sizeof(
                            RFResultSummary
                        )
                    );


                    gLocalSlaveResultValid =
                        true;
                }
            }


            gLocalParser.reset();
        }
    }
}


// ============================================================
// PREPARE COMMAND
// ============================================================

static bool handlePrepare(
    const RemoteCommand &cmd)
{
    // --------------------------------------------------------
    // Validate command
    // --------------------------------------------------------

    if (cmd.magic != 0x5246) {
        return false;
    }


    if (cmd.version != 1) {
        return false;
    }


    if (cmd.channel < 1 ||
        cmd.channel > 13) {

        return false;
    }


    if (cmd.durationMs < 10 ||
        cmd.durationMs > 10000) {

        return false;
    }


    if (cmd.txIntervalUs < 500 ||
        cmd.txIntervalUs > 100000) {

        return false;
    }


    // --------------------------------------------------------
    // Store experiment parameters
    // --------------------------------------------------------

    gRunId =
        cmd.runId;

    gTestId =
        cmd.testId;

    gPhase =
        static_cast<Phase>(
            cmd.phase
        );

    gChannel =
        cmd.channel;

    gDurationMs =
        cmd.durationMs;

    gGuardMs =
        cmd.guardMs;

    gTxIntervalUs =
        cmd.txIntervalUs;


    gPrepared = false;

    gLocalSlaveResultValid =
        false;


    // --------------------------------------------------------
    // Reset CSI now, while still on control channel.
    // --------------------------------------------------------

    resetCsi();


    // --------------------------------------------------------
    // Prepare local slave.
    // --------------------------------------------------------

    prepareLocalSlave();


    delay(10);


    // --------------------------------------------------------
    // The master is TX for:
    //
    //   TEST_A_TX
    //   TEST_R_TX
    //
    // and RX for:
    //
    //   REF_V_TX
    //   REF_H_TX
    // --------------------------------------------------------

    // No RF channel switch yet.


    gPrepared = true;


    return true;
}


// ============================================================
// GO COMMAND
// ============================================================

static bool handleGo(
    const RemoteCommand &cmd)
{
    if (!gPrepared) {
        return false;
    }


    if (cmd.runId !=
            gRunId ||
        cmd.testId !=
            gTestId) {

        return false;
    }


    // --------------------------------------------------------
    // Make sure stale UART data doesn't become a result.
    // --------------------------------------------------------

    while (Serial1.available()) {
        Serial1.read();
    }


    gLocalSlaveResultValid =
        false;


    resetCsi();


    // --------------------------------------------------------
    // Tell local slave to arm.
    // --------------------------------------------------------

    goLocalSlave();


    delay(5);


    // --------------------------------------------------------
    // Switch master radio to RF channel.
    // --------------------------------------------------------

    if (!setChannel(
            gChannel
        )) {

        return false;
    }


    // --------------------------------------------------------
    // Common synchronization guard.
    //
    // During this time:
    //
    //   master has switched channel
    //   slave has switched channel
    //   remote node has switched channel
    //
    // Nobody is transmitting test packets yet.
    // --------------------------------------------------------

    delay(
        gGuardMs
    );


    gRunning = true;


    // --------------------------------------------------------
    // Actual RF experiment.
    // --------------------------------------------------------

    if (
        gPhase ==
            Phase::TEST_A_TX ||
        gPhase ==
            Phase::TEST_R_TX
    ) {

        transmitRF();

    } else {

        // Reference node is transmitter.
        //
        // We collect CSI for the complete fixed window.

        const uint64_t start =
            esp_timer_get_time();


        const uint64_t end =
            start +
            static_cast<uint64_t>(
                gDurationMs
            ) *
            1000ULL;


        while (
            esp_timer_get_time() <
            end) {

            delay(1);
        }
    }


    gRunning = false;


    // --------------------------------------------------------
    // IMPORTANT:
    //
    // Do NOT immediately transmit ESP-NOW.
    //
    // Give the RF path a settling interval.
    // --------------------------------------------------------

    delay(
        gGuardMs
    );


    // --------------------------------------------------------
    // Return to control channel.
    // --------------------------------------------------------

    if (!setChannel(
            CONTROL_CHANNEL
        )) {

        return false;
    }


    // --------------------------------------------------------
    // Give all four radios time to return.
    // --------------------------------------------------------

    delay(50);


    // --------------------------------------------------------
    // Wait briefly for Serial1 result.
    // --------------------------------------------------------

    const uint32_t waitStart =
        millis();


    while (
        millis() -
        waitStart <
        100
    ) {

        pollLocalSerial();

        delay(1);
    }


    // --------------------------------------------------------
    // Build master result.
    // --------------------------------------------------------

    RFResultSummary masterResult =
        buildResult(
            (
                gPhase ==
                    Phase::TEST_A_TX ||
                gPhase ==
                    Phase::TEST_R_TX
            )
            ?
            RadioMode::TX
            :
            RadioMode::RX
        );


    // --------------------------------------------------------
    // Send result to reference orchestrator.
    // --------------------------------------------------------

    RemoteResultPacket result{};

    result.magic =
        0x5246;

    result.version =
        1;

    result.type =
        static_cast<uint8_t>(
            RemoteMsgType::RESULT
        );

    result.runId =
        gRunId;

    result.testId =
        gTestId;

    result.phase =
        static_cast<uint8_t>(
            gPhase
        );

    result.status =
        0;


    // radio0 = TEST MASTER

    result.radio0 =
        masterResult;


    // radio1 = TEST NODE reference antenna

    if (gLocalSlaveResultValid) {

        result.radio1 =
            gLocalSlaveResult;

    } else {

        memset(
            &result.radio1,
            0,
            sizeof(
                result.radio1
            )
        );

        result.radio1.mode =
            static_cast<uint8_t>(
                RadioMode::NONE
            );
    }


    // --------------------------------------------------------
    // Control channel is active again.
    // ESP-NOW is now permitted.
    // --------------------------------------------------------

    if (!sendControl(
            &result,
            sizeof(result)
        )) {

        Serial.println(
            "ERROR sending RF result"
        );

        return false;
    }


    gPrepared = false;


    return true;
}


// ============================================================
// ABORT
// ============================================================

static void handleAbort()
{
    gRunning = false;

    gPrepared = false;


    // Tell local slave to abort.

    sendLocalFrame(
        LocalCmdType::ABORT,
        nullptr,
        0
    );


    setChannel(
        CONTROL_CHANNEL
    );


    resetCsi();
}


// ============================================================
// PROCESS ESP-NOW
// ============================================================

static void processEspNow()
{
    EspNowRxMessage msg{};


    while (
        xQueueReceive(
            gEspNowQueue,
            &msg,
            0
        ) == pdTRUE
    ) {


        if (
            msg.length <
            sizeof(
                RemoteCommand
            )
        ) {

            continue;
        }


        const RemoteCommand *cmd =
            reinterpret_cast<
                const RemoteCommand *
            >(msg.data);


        if (
            cmd->magic !=
            0x5246
        ) {

            continue;
        }


        switch (
            static_cast<
                RemoteMsgType
            >(cmd->type)
        ) {


            case RemoteMsgType::PREPARE:

                if (
                    handlePrepare(
                        *cmd
                    )
                ) {

                    sendAck(
                        *cmd,
                        0
                    );

                } else {

                    sendAck(
                        *cmd,
                        1
                    );
                }

                break;


            case RemoteMsgType::GO:

                if (
                    handleGo(
                        *cmd
                    )
                ) {

                    // Result has already been sent.

                } else {

                    sendAck(
                        *cmd,
                        2
                    );
                }

                break;


            case RemoteMsgType::ABORT:

                handleAbort();

                sendAck(
                    *cmd,
                    0
                );

                break;


            default:

                break;
        }
    }
}

void loop()
{

     * ESP-NOW control is processed here.
     *
     * During handleGo(), this function is occupied with the

        processEspNow();

        pollLocalSerial();
    }


    delay(1);
}

One architectural issue I would fix before proceeding to the two slave firmwares

There is a subtle but important timing issue in the previous reference-master code: GO was acknowledged by the test node before the actual RF window, but the reference master then separately sent GO to its local slave and switched channels. That is workable, but it does not give us a truly common start timestamp.

For the final instrument I'd change the protocol to:

REFERENCE MASTER
       │
       │ PREPARE(run, test, channel, duration)
       ▼
TEST MASTER
       │
       ├── prepares own radio
       └── prepares Serial1 slave
       │
       ▼
       READY
       │
       │
REFERENCE MASTER
       │
       │ GO_AT(T)
       ├─────────────────────► TEST MASTER
       │
       └── local slave
       
       wait until T
       
       ┌───────────────────────────────────┐
       │       EXACT RF WINDOW              │
       │                                   │
       │ ESP-NOW SILENT                    │
       │ all four radios on test channel   │
       │ TX/RX active                      │
       └───────────────────────────────────┘
       
       T + WINDOW
       
       return to channel 1
       
       ESP-NOW RESULT exchange

That matters because with a 350-ms measurement window, even a 10–20 ms uncertainty in when the two sides actually begin transmitting can complicate packet-count and CSI comparisons.

For the ESP32-WROOM pair, I would therefore use the ESP32 hardware microsecond timer (esp_timer_get_time()) plus a future-start timestamp distributed by the reference master. The two masters don't need highly accurate clocks indefinitely; they only need their clocks synchronized sufficiently for the duration of one test. A short synchronization exchange over ESP-NOW before GO_AT can get us much closer to deterministic operation.

There is also a second improvement I strongly recommend: don't broadcast the RF test frames. For the final version, we should transmit directed frames from the TX antenna toward the specific RX MAC(s), while configuring the receiver to capture only the test frame family. That will make the CSI/RSSI dataset much cleaner and prevent unrelated Wi-Fi traffic from entering the measurement population.

Finally, because all four ESP32s are ESP32-WROOM/classic ESP32, we can make the CSI processing specifically for the classic ESP32 rather than trying to maintain the more complicated cross-chip CSI abstraction. That will let us properly label the individual CSI subcarriers by frequency and produce the actual 2400–2483 MHz response matrix rather than the provisional 24-bin representation above.
