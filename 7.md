Below is the test-node slave firmware for the ESP32-WROOM. It controls the reference ¼-wave antenna on the test node, attached through Serial1 to the test-node master.

Its behavior is symmetric with the reference-node slave:

TEST NODE

ESP32 MASTER                    ESP32 SLAVE
test antenna                   reference antenna
     │                               │
     └──────────── Serial1 ──────────┘

The master tells it which phase is running. The slave determines whether its own antenna is transmitting only during TEST_R_TX; otherwise it is an RX/CSI instrument.

/*
 * ============================================================
 * QUAD ESP32 RF ANTENNA TESTER
 *
 * TEST NODE SLAVE
 *
 * ESP32-WROOM
 *
 * ROLE:
 *   Reference 1/4-wave antenna on the TEST NODE
 *
 * CONNECTION:
 *
 *       TEST NODE MASTER
 *       TEST ANTENNA
 *             |
 *           Serial1
 *             |
 *       TEST NODE SLAVE
 *       REFERENCE ANTENNA
 *
 * ============================================================
 *
 * Communication:
 *
 *   Serial1 only
 *
 * RF:
 *
 *   ESP-IDF Wi-Fi driver
 *   promiscuous RX
 *   CSI
 *   esp_wifi_80211_tx()
 *
 * During RF measurement:
 *
 *   NO Serial1
 *   NO ESP-NOW
 *   NO control traffic
 *
 * ============================================================
 */

struct CsiAccumulator {

    volatile uint32_t frames;

    volatile int32_t rssiSum;

    volatile uint32_t rssiSumSq;

    volatile int8_t rssiMin;

    volatile int8_t rssiMax;

    volatile uint32_t pairCountTotal;

    int32_t sumI[MAX_CSI_PAIRS];

    int32_t sumQ[MAX_CSI_PAIRS];

    uint16_t count[MAX_CSI_PAIRS];
};


static CsiAccumulator gCSI;

static portMUX_TYPE gCSIMux =
    portMUX_INITIALIZER_UNLOCKED;

struct TestPacketStats {

    volatile uint32_t packets;

    volatile uint32_t matchingPackets;

    volatile uint32_t lastSequence;

    volatile bool haveSequence;
};


static TestPacketStats gPackets;


// ============================================================
// RESET
// ============================================================

static void resetMeasurements()
{
    portENTER_CRITICAL(
        &gCSIMux
    );

    memset(
        &gCSI,
        0,
        sizeof(gCSI)
    );

    gCSI.rssiMin = 127;

    gCSI.rssiMax = -127;

    memset(
        &gPackets,
        0,
        sizeof(gPackets)
    );

    portEXIT_CRITICAL(
        &gCSIMux
    );
}


// ============================================================
// CSI CALLBACK
// ============================================================

static void IRAM_ATTR csiCallback(
    void *ctx,
    wifi_csi_info_t *data)
{
    if (data == nullptr) {
        return;
    }

    if (data->buf == nullptr) {
        return;
    }

    if (data->len < 2) {
        return;
    }


    const int pairs =
        min(
            data->len / 2,
            static_cast<int>(
                MAX_CSI_PAIRS
            )
        );


    const int8_t rssi =
        data->rx_ctrl.rssi;


    portENTER_CRITICAL_ISR(
        &gCSIMux
    );


    ++gCSI.frames;


    gCSI.rssiSum += rssi;


    gCSI.rssiSumSq +=
        static_cast<uint32_t>(
            rssi * rssi
        );


    if (rssi < gCSI.rssiMin) {
        gCSI.rssiMin = rssi;
    }


    if (rssi > gCSI.rssiMax) {
        gCSI.rssiMax = rssi;
    }


    gCSI.pairCountTotal +=
        pairs;


    for (
        int i = 0;
        i < pairs;
        ++i
    ) {

        /*
         * Signed complex CSI byte pairs.
         *
         * Absolute subcarrier mapping is intentionally left
         * separate from this hardware-collection layer.
         */

        const int8_t imag =
            data->buf[
                2 * i
            ];


        const int8_t real =
            data->buf[
                2 * i + 1
            ];


        gCSI.sumI[i] +=
            real;


        gCSI.sumQ[i] +=
            imag;


        ++gCSI.count[i];
    }


    portEXIT_CRITICAL_ISR(
        &gCSIMux
    );
}


// ============================================================
// PROMISCUOUS PACKET FILTER
// ============================================================

static void promiscuousCallback(
    void *packet,
    wifi_promiscuous_pkt_type_t type)
{
    if (packet == nullptr) {
        return;
    }


    if (
        type != WIFI_PKT_MGMT &&
        type != WIFI_PKT_DATA
    ) {

        return;
    }


    wifi_promiscuous_pkt_t *pkt =
        static_cast<
            wifi_promiscuous_pkt_t *
        >(packet);


    const uint16_t len =
        pkt->rx_ctrl.sig_len;


    if (len < 46) {
        return;
    }


    const uint8_t *frame =
        pkt->payload;


    const uint16_t fc =
        frame[0] |
        (
            static_cast<uint16_t>(
                frame[1]
            ) << 8
        );


    const uint8_t frameType =
        (fc >> 2) & 0x03;


    const uint8_t subtype =
        (fc >> 4) & 0x0F;


    /*
     * Action frame.
     */

    if (
        frameType != 0 ||
        subtype != 13
    ) {

        return;
    }


    /*
     * Test protocol:
     *
     * byte 24: category
     * byte 25-27: DE AD BE
     * byte 28: 02
     * byte 29-32: RFTS
     */

    if (frame[24] != 127) {
        return;
    }


    if (
        frame[25] != 0xDE ||
        frame[26] != 0xAD ||
        frame[27] != 0xBE
    ) {

        return;
    }


    if (frame[28] != 0x02) {
        return;
    }


    if (
        frame[29] != 0x52 ||
        frame[30] != 0x46 ||
        frame[31] != 0x54 ||
        frame[32] != 0x53
    ) {

        return;
    }


    /*
     * Check run/test identifiers.
     */

    uint32_t runId =
        frame[33] |
        (
            static_cast<uint32_t>(
                frame[34]
            ) << 8
        ) |
        (
            static_cast<uint32_t>(
                frame[35]
            ) << 16
        ) |
        (
            static_cast<uint32_t>(
                frame[36]
            ) << 24
        );


    uint16_t testId =
        frame[37] |
        (
            static_cast<uint16_t>(
                frame[38]
            ) << 8
        );


    if (
        runId != gRunId ||
        testId != gTestId
    ) {

        return;
    }


    /*
     * Sequence field:
     *
     * 40..43
     */

    const uint32_t sequence =
        frame[40] |
        (
            static_cast<uint32_t>(
                frame[41]
            ) << 8
        ) |
        (
            static_cast<uint32_t>(
                frame[42]
            ) << 16
        ) |
        (
            static_cast<uint32_t>(
                frame[43]
            ) << 24
        );


    portENTER_CRITICAL_ISR(
        &gCSIMux
    );


    ++gPackets.packets;

    ++gPackets.matchingPackets;


    if (
        gPackets.haveSequence &&
        sequence <=
            gPackets.lastSequence
    ) {

        ++gPackets.duplicates;
    }


    gPackets.lastSequence =
        sequence;

    gPackets.haveSequence =
        true;


    portEXIT_CRITICAL_ISR(
        &gCSIMux
    );
}


// ============================================================
// WIFI INITIALIZATION
// ============================================================

static void initWiFi()
{
    WiFi.mode(
        WIFI_STA
    );


    WiFi.disconnect(
        true,
        true
    );


    delay(50);


    esp_err_t err;


    err =
        esp_wifi_set_ps(
            WIFI_PS_NONE
        );


    if (err != ESP_OK) {

        Serial.printf(
            "WARNING power save: %s\n",
            esp_err_to_name(err)
        );
    }


    err =
        esp_wifi_set_channel(
            CONTROL_CHANNEL,
            WIFI_SECOND_CHAN_NONE
        );


    if (err != ESP_OK) {

        Serial.printf(
            "FATAL channel: %s\n",
            esp_err_to_name(err)
        );

        while (true) {
            delay(1000);
        }
    }


    err =
        esp_wifi_set_promiscuous(
            true
        );


    if (err != ESP_OK) {

        Serial.printf(
            "FATAL promiscuous: %s\n",
            esp_err_to_name(err)
        );

        while (true) {
            delay(1000);
        }
    }


    err =
        esp_wifi_set_promiscuous_rx_cb(
            promiscuousCallback
        );


    if (err != ESP_OK) {

        Serial.printf(
            "FATAL promisc callback: %s\n",
            esp_err_to_name(err)
        );

        while (true) {
            delay(1000);
        }
    }


    // --------------------------------------------------------
    // CSI
    // --------------------------------------------------------

    err =
        esp_wifi_set_csi_rx_cb(
            csiCallback,
            nullptr
        );


    if (err != ESP_OK) {

        Serial.printf(
            "FATAL CSI callback: %s\n",
            esp_err_to_name(err)
        );

        while (true) {
            delay(1000);
        }
    }


    wifi_csi_config_t csiConfig{};


    csiConfig.lltf_en =
        true;

    csiConfig.htltf_en =
        true;

    csiConfig.stbc_htltf2_en =
        true;

    csiConfig.ltf_merge_en =
        false;

    csiConfig.channel_filter_en =
        false;

    csiConfig.manu_scale =
        false;

    csiConfig.shift =
        0;


    err =
        esp_wifi_set_csi_config(
            &csiConfig
        );


    if (err != ESP_OK) {

        Serial.printf(
            "FATAL CSI config: %s\n",
            esp_err_to_name(err)
        );

        while (true) {
            delay(1000);
        }
    }


    err =
        esp_wifi_set_csi(
            true
        );


    if (err != ESP_OK) {

        Serial.printf(
            "FATAL CSI enable: %s\n",
            esp_err_to_name(err)
        );

        while (true) {
            delay(1000);
        }
    }


    /*
     * Fixed TX power.
     */

    err =
        esp_wifi_set_max_tx_power(
            78
        );


    if (err != ESP_OK) {

        Serial.printf(
            "WARNING TX power: %s\n",
            esp_err_to_name(err)
        );
    }
}

// SET CHANNEL
// ============================================================

static bool setChannel(
    uint8_t channel)
{
    const esp_err_t err =
        esp_wifi_set_channel(
            channel,
            WIFI_SECOND_CHAN_NONE
        );


    if (err != ESP_OK) {

        Serial.printf(
            "channel %u failed: %s\n",
            channel,
            esp_err_to_name(err)
        );

        return false;
    }


    return true;
}


// ============================================================
// CRC16
// ============================================================

static uint16_t crc16(
    const uint8_t *data,
    size_t length)
{
    uint16_t crc =
        0xFFFF;


    for (
        size_t i = 0;
        i < length;
        ++i
    ) {

        crc ^=
            data[i];


        for (
            uint8_t j = 0;
            j < 8;
            ++j
        ) {

            if (crc & 1) {

                crc =
                    (crc >> 1) ^
                    0xA001;

            } else {

                crc >>= 1;
            }
        }
    }


    return crc;
}


// ============================================================
// SERIAL PARSER
// ============================================================

class SerialParser {

public:

    enum class State : uint8_t {

        MAGIC1,

        MAGIC2,

        TYPE,

        LENGTH,

        PAYLOAD,

        CRC_LO,

        CRC_HI
    };


    State state =
        State::MAGIC1;


    uint8_t type = 0;

    uint8_t length = 0;

    uint8_t payload[
        SERIAL_MAX_PAYLOAD
    ];

    uint8_t position = 0;

    uint16_t receivedCRC = 0;


    void reset()
    {
        state =
            State::MAGIC1;

        type = 0;

        length = 0;

        position = 0;

        receivedCRC = 0;
    }


    bool feed(
        uint8_t value)
    {
        switch (state) {


            case State::MAGIC1:

                if (
                    value ==
                    SERIAL_MAGIC1
                ) {

                    state =
                        State::MAGIC2;
                }

                break;


            case State::MAGIC2:

                if (
                    value ==
                    SERIAL_MAGIC2
                ) {

                    state =
                        State::TYPE;

                } else {

                    reset();
                }

                break;


            case State::TYPE:

                type =
                    value;

                state =
                    State::LENGTH;

                break;


            case State::LENGTH:

                length =
                    value;


                if (
                    length >
                    SERIAL_MAX_PAYLOAD
                ) {

                    reset();

                } else if (
                    length == 0
                ) {

                    state =
                        State::CRC_LO;

                } else {

                    position = 0;

                    state =
                        State::PAYLOAD;
                }

                break;


            case State::PAYLOAD:

                payload[
                    position++
                ] =
                    value;


                if (
                    position >=
                    length
                ) {

                    state =
                        State::CRC_LO;
                }

                break;


            case State::CRC_LO:

                receivedCRC =
                    value;

                state =
                    State::CRC_HI;

                break;


            case State::CRC_HI:

                receivedCRC |=
                    static_cast<
                        uint16_t
                    >(value) << 8;


                {

                    SerialHeader header{};


                    header.magic1 =
                        SERIAL_MAGIC1;


                    header.magic2 =
                        SERIAL_MAGIC2;


                    header.type =
                        type;


                    header.length =
                        length;


                    uint16_t calculated =
                        crc16(
                            reinterpret_cast<
                                const uint8_t *
                            >(
                                &header
                            ),
                            sizeof(header)
                        );


                    for (
                        uint8_t i = 0;
                        i < length;
                        ++i
                    ) {

                        uint8_t b =
                            payload[i];


                        calculated ^=
                            b;


                        for (
                            uint8_t j = 0;
                            j < 8;
                            ++j
                        ) {

                            if (
                                calculated & 1
                            ) {

                                calculated =
                                    (
                                        calculated
                                        >> 1
                                    ) ^
                                    0xA001;

                            } else {

                                calculated >>=
                                    1;
                            }
                        }
                    }


                    if (
                        calculated ==
                        receivedCRC
                    ) {

                        state =
                            State::MAGIC1;

                        return true;
                    }
                }


                reset();

                break;
        }


        return false;
    }
};


static SerialParser gParser;


// ============================================================
// SEND SERIAL FRAME
// ============================================================

static void sendSerialFrame(
    LocalType type,
    const void *payload,
    uint8_t length)
{
    SerialHeader header{};


    header.magic1 =
        SERIAL_MAGIC1;

    header.magic2 =
        SERIAL_MAGIC2;

    header.type =
        static_cast<
            uint8_t
        >(type);

    header.length =
        length;


    Serial1.write(
        reinterpret_cast<
            const uint8_t *
        >(
            &header
        ),
        sizeof(header)
    );


    if (
        payload &&
        length
    ) {

        Serial1.write(
            static_cast<
                const uint8_t *
            >(
                payload
            ),
            length
        );
    }


    uint16_t crc =
        crc16(
            reinterpret_cast<
                const uint8_t *
            >(
                &header
            ),
            sizeof(header)
        );


    if (
        payload &&
        length
    ) {

        const uint8_t *p =
            static_cast<
                const uint8_t *
            >(
                payload
            );


        for (
            uint8_t i = 0;
            i < length;
            ++i
        ) {

            uint8_t b =
                p[i];


            crc ^= b;


            for (
                uint8_t j = 0;
                j < 8;
                ++j
            ) {

                if (crc & 1) {

                    crc =
                        (crc >> 1) ^
                        0xA001;

                } else {

                    crc >>= 1;
                }
            }
        }
    }


    Serial1.write(
        reinterpret_cast<
            const uint8_t *
        >(
            &crc
        ),
        sizeof(crc)
    );


    Serial1.flush();
}


// ============================================================
// HANDLE PREPARE
// ============================================================

static bool handlePrepare(
    const PrepareCommand &cmd)
{
    if (
        cmd.magic != RF_MAGIC ||
        cmd.version != PROTOCOL_VERSION ||
        cmd.type != static_cast<
            uint8_t
        >(LocalType::PREPARE)
    ) {

        return false;
    }


    if (
        cmd.channel < 1 ||
        cmd.channel > 13
    ) {

        return false;
    }


    if (
        cmd.windowMs < 50 ||
        cmd.windowMs > 10000
    ) {

        return false;
    }


    gRunId =
        cmd.runId;


    gTestId =
        cmd.testId;


    gPhase =
        static_cast<
            Phase
        >(cmd.phase);


    gChannel =
        cmd.channel;


    gWindowMs =
        cmd.windowMs;


    gGuardUs =
        cmd.guardUs;


    gTxIntervalUs =
        cmd.txIntervalUs;


    gBandwidth =
        cmd.bandwidth;


    gTxRate =
        cmd.txRate;


    resetMeasurements();


    /*
     * The test-node reference antenna transmits only during
     * TEST_R_TX.
     */

    gPrepared =
        true;


    /*
     * Remain on the control channel until START_AT arrives.
     */

    return setChannel(
        CONTROL_CHANNEL
    );
}


// ============================================================
// SEND READY
// ============================================================

static void sendReady()
{
    ReadyMessage ready{};


    ready.magic =
        RF_MAGIC;

    ready.version =
        PROTOCOL_VERSION;

    ready.type =
        static_cast<
            uint8_t
        >(LocalType::READY);

    ready.runId =
        gRunId;

    ready.testId =
        gTestId;

    ready.status =
        0;


    sendSerialFrame(
        LocalType::READY,
        &ready,
        sizeof(ready)
    );
}


// ============================================================
// WAIT UNTIL START
// ============================================================

static void waitUntil(
    uint64_t timestampUs)
{
    while (true) {

        const uint64_t now =
            esp_timer_get_time();


        if (
            now >=
            timestampUs
        ) {

            return;
        }


        const uint64_t remaining =
            timestampUs -
            now;


        if (
            remaining >
            5000
        ) {

            delay(
                static_cast<
                    uint32_t
                >(
                    remaining /
                    1000
                )
            );

        } else {

            delayMicroseconds(
                static_cast<
                    uint32_t
                >(
                    remaining
                )
            );
        }
    }
}

// BUILD RF TEST PACKET
// ============================================================

static uint16_t buildReferencePacket(
    uint8_t *frame,
    uint32_t sequence)
{
    uint16_t p = 0;


    /*
     * Management Action frame.
     */

    const uint16_t fc =
        0x00D0;


    memcpy(
        frame + p,
        &fc,
        2
    );

    p += 2;


    uint16_t duration = 0;


    memcpy(
        frame + p,
        &duration,
        2
    );

    p += 2;


    /*
     * Destination is filled as broadcast here because the
     * slave does not own the global test configuration.
     *
     * For the final integrated version, the master should
     * provide the receiving reference-node MAC in PREPARE.
     */

    memset(
        frame + p,
        0xFF,
        6
    );

    p += 6;


    uint8_t mac[6];


    esp_wifi_get_mac(
        WIFI_IF_STA,
        mac
    );


    memcpy(
        frame + p,
        mac,
        6
    );

    p += 6;


    memset(
        frame + p,
        0xFF,
        6
    );

    p += 6;


    uint16_t seqControl = 0;


    memcpy(
        frame + p,
        &seqControl,
        2
    );

    p += 2;


    frame[p++] = 127;


    frame[p++] = 0xDE;
    frame[p++] = 0xAD;
    frame[p++] = 0xBE;


    frame[p++] = 0x02;


    frame[p++] = 0x52;
    frame[p++] = 0x46;
    frame[p++] = 0x54;
    frame[p++] = 0x53;


    memcpy(
        frame + p,
        &gRunId,
        4
    );

    p += 4;


    memcpy(
        frame + p,
        &gTestId,
        2
    );

    p += 2;


    frame[p++] =
        static_cast<
            uint8_t
        >(gPhase);


    memcpy(
        frame + p,
        &sequence,
        4
    );

    p += 4;


    frame[p++] =
        gChannel;


    /*
     * Identify this as the test-node reference antenna.
     * ASCII 'R' is used merely as a binary identifier.
     */

    frame[p++] = 0x52;


    while (
        p < 90
    ) {

        frame[p++] =
            static_cast<
                uint8_t
            >(
                0xA5 ^
                (
                    sequence &
                    0xFF
                )
            );
    }


    return p;
}


// ============================================================
// TRANSMIT WINDOW
// ============================================================

static void performTransmit(
    uint64_t startAtUs,
    uint32_t windowUs)
{
    waitUntil(
        startAtUs
    );


    const uint64_t endAtUs =
        startAtUs +
        windowUs;


    uint64_t nextTx =
        startAtUs;


    uint32_t sequence = 0;


    uint8_t frame[128];


    while (
        esp_timer_get_time() <
        endAtUs
    ) {


        const uint64_t now =
            esp_timer_get_time();


        if (
            now <
            nextTx
        ) {

            const uint64_t waitUs =
                nextTx -
                now;


            if (
                waitUs > 1000
            ) {

                delay(
                    static_cast<
                        uint32_t
                    >(
                        waitUs /
                        1000
                    )
                );

            } else {

                delayMicroseconds(
                    static_cast<
                        uint32_t
                    >(
                        waitUs
                    )
                );
            }


            continue;
        }


        const uint16_t length =
            buildReferencePacket(
                frame,
                sequence
            );


        (void)
            esp_wifi_80211_tx(
                WIFI_IF_STA,
                frame,
                length,
                true
            );


        ++sequence;


        nextTx +=
            gTxIntervalUs;
    }
}


// ============================================================
// RECEIVE WINDOW
// ============================================================

static void performReceive(
    uint64_t startAtUs,
    uint32_t windowUs)
{
    waitUntil(
        startAtUs
    );


    const uint64_t endAtUs =
        startAtUs +
        windowUs;


    while (
        esp_timer_get_time() <
        endAtUs
    ) {

        /*
         * CSI and packet callbacks handle RX.
         *
         * No Serial1 communication here.
         */

        delay(1);
    }
}


// ============================================================
// BUILD RESULT
// ============================================================

static RFResultSummary buildResult(
    RadioMode mode)
{
    RFResultSummary result{};


    result.mode =
        static_cast<
            uint8_t
        >(mode);


    portENTER_CRITICAL(
        &gCSIMux
    );


    result.csiFrames =
        static_cast<
            uint16_t
        >(
            min<uint32_t>(
                gCSI.frames,
                UINT16_MAX
            )
        );


    result.csiValid =
        gCSI.frames >
        0;


    result.testPackets =
        gPackets.packets;


    result.matchingPackets =
        gPackets.matchingPackets;


    result.crcPackets =
        gPackets.crcPackets;


    result.duplicatePackets =
        gPackets.duplicates;


    result.rssiSum =
        gCSI.rssiSum;


    result.rssiSumSq =
        gCSI.rssiSumSq;


    result.rssiMin =
        gCSI.rssiMin;


    result.rssiMax =
        gCSI.rssiMax;


    uint16_t pairs = 0;


    if (
        gCSI.frames
    ) {

        pairs =
            static_cast<
                uint16_t
            >(
                min<uint32_t>(
                    MAX_CSI_PAIRS,
                    gCSI.pairCountTotal /
                    gCSI.frames
                )
            );
    }


    result.csiPairs =
        pairs;


    for (
        uint8_t out = 0;
        out < EXPORT_CSI_BINS;
        ++out
    ) {


        uint16_t source = 0;


        if (
            pairs > 1
        ) {

            source =
                static_cast<
                    uint16_t
                >(
                    (
                        static_cast<
                            uint32_t
                        >(out) *
                        (
                            pairs - 1
                        )
                    ) /
                    (
                        EXPORT_CSI_BINS -
                        1
                    )
                );
        }


        const uint16_t n =
            gCSI.count[
                source
            ];


        if (n) {

            result.avgI[out] =
                static_cast<
                    int8_t
                >(
                    gCSI.sumI[
                        source
                    ] /
                    static_cast<
                        int32_t
                    >(n)
                );


            result.avgQ[out] =
                static_cast<
                    int8_t
                >(
                    gCSI.sumQ[
                        source
                    ] /
                    static_cast<
                        int32_t
                    >(n)
                );

        } else {

            result.avgI[out] = 0;

            result.avgQ[out] = 0;
        }
    }


    portEXIT_CRITICAL(
        &gCSIMux
    );


    return result;
}


// ============================================================
// SEND RESULT
// ============================================================

static void sendResult()
{
    const bool transmit =
        (
            gPhase ==
            Phase::TEST_R_TX
        );


    RFResultSummary result =
        buildResult(
            transmit
            ?
            RadioMode::TX
            :
            RadioMode::RX
        );


    sendSerialFrame(
        LocalType::RESULT,
        &result,
        sizeof(result)
    );
}


// ============================================================
// START AT
// ============================================================

static bool handleStartAt(
    const StartAtCommand &cmd)
{
    if (!gPrepared) {
        return false;
    }


    if (
        cmd.magic !=
            RF_MAGIC ||
        cmd.version !=
            PROTOCOL_VERSION ||
        cmd.type !=
            static_cast<
                uint8_t
            >(LocalType::START_AT)
    ) {

        return false;
    }


    if (
        cmd.runId !=
            gRunId ||
        cmd.testId !=
            gTestId
    ) {

        return false;
    }


    /*
     * Switch RF channel before the scheduled start.
     */

    if (
        !setChannel(
            gChannel
        )
    ) {

        return false;
    }


    resetMeasurements();


    gRunning = true;


    const bool transmit =
        (
            gPhase ==
            Phase::TEST_R_TX
        );


    if (transmit) {

        performTransmit(
            cmd.startAtUs,
            cmd.windowUs
        );

    } else {

        performReceive(
            cmd.startAtUs,
            cmd.windowUs
        );
    }


    /*
     * Do not return to Serial1 yet.
     */

    const uint64_t returnAt =
        cmd.startAtUs +
        cmd.windowUs +
        cmd.guardAfterUs;


    while (
        esp_timer_get_time() <
        returnAt
    ) {

        delay(1);
    }


    if (
        !setChannel(
            CONTROL_CHANNEL
        )
    ) {

        gRunning = false;

        return false;
    }


    gRunning = false;


    delay(5);


    /*
     * Result goes back to TEST NODE MASTER.
     */

    sendResult();


    gPrepared = false;


    return true;
}


// ============================================================
// SYNC REQUEST
// ============================================================

static void handleSyncRequest(
    const SyncRequest &req)
{
    if (
        req.magic !=
            RF_MAGIC ||
        req.version !=
            PROTOCOL_VERSION ||
        req.type !=
            static_cast<
                uint8_t
            >(LocalType::SYNC_REQ)
    ) {

        return;
    }


    const uint64_t t2 =
        esp_timer_get_time();


    SyncResponse response{};


    response.magic =
        RF_MAGIC;


    response.version =
        PROTOCOL_VERSION;


    response.type =
        static_cast<
            uint8_t
        >(LocalType::SYNC_RESP);


    response.sequence =
        req.sequence;


    response.t1 =
        req.t1;


    response.t2 =
        t2;


    response.t3 =
        esp_timer_get_time();


    sendSerialFrame(
        LocalType::SYNC_RESP,
        &response,
        sizeof(response)
    );
}


// ============================================================
// ABORT
// ============================================================

static void handleAbort()
{
    gRunning = false;

    gPrepared = false;


    resetMeasurements();


    setChannel(
        CONTROL_CHANNEL
    );
}


// ============================================================
// PROCESS FRAME
// ============================================================

static void processFrame()
{
    const LocalType type =
        static_cast<
            LocalType
        >(gParser.type);


    switch (type) {


        case LocalType::PREPARE:

            if (
                gParser.length >=
                sizeof(
                    PrepareCommand
                )
            ) {

                PrepareCommand cmd{};


                memcpy(
                    &cmd,
                    gParser.payload,
                    sizeof(cmd)
                );


                if (
                    handlePrepare(
                        cmd
                    )
                ) {

                    sendReady();
                }
            }

            break;


        case LocalType::START_AT:

            if (
                gParser.length >=
                sizeof(
                    StartAtCommand
                )
            ) {

                StartAtCommand cmd{};


                memcpy(
                    &cmd,
                    gParser.payload,
                    sizeof(cmd)
                );


                handleStartAt(
                    cmd
                );
            }

            break;


        case LocalType::SYNC_REQ:

            if (
                gParser.length >=
                sizeof(
                    SyncRequest
                )
            ) {

                SyncRequest req{};


                memcpy(
                    &req,
                    gParser.payload,
                    sizeof(req)
                );


                handleSyncRequest(
                    req
                );
            }

            break;


        case LocalType::ABORT:

            handleAbort();

            break;


        default:

            break;
    }
}


// ============================================================
// SERIAL POLLER
// ============================================================

static void pollSerial()
{
    /*
     * Absolutely no Serial1 processing during the RF window.
     */

    if (gRunning) {
        return;
    }


    while (
        Serial1.available()
    ) {

        const int v =
            Serial1.read();


        if (v < 0) {
            break;
        }


        if (
            gParser.feed(
                static_cast<
                    uint8_t
                >(v)
            )
        ) {

            processFrame();

            gParser.reset();
        }
    }
}

There is one important integration issue in the four firmware pieces that I would resolve before treating the system as a finished instrument: the two slave transmitters currently use broadcast destination addresses because the destination MAC is not yet part of PrepareCommand. For the experiment you described, I would make that a first-class field.

The final PrepareCommand should contain:

uint8_t txDestinationMac[6];
uint8_t txSourceRole;
uint8_t rxRole;

so every transmitter knows exactly which master is its intended receiver. Then the packet matrix becomes unambiguous:

REF-V  ───────► TEST-MASTER
REF-V  ───────► TEST-SLAVE

REF-H  ───────► TEST-MASTER
REF-H  ───────► TEST-SLAVE

TEST-A ───────► REF-MASTER
TEST-A ───────► REF-SLAVE

TEST-R ───────► REF-MASTER
TEST-R ───────► REF-SLAVE

That also gives us a cleaner way to correlate the CSI callback with the intended transmitter.

One more correction worth making before final integration: the sequence-number offsets in the raw-frame parsers should be centralized in one shared protocol header. Right now the four sketches duplicate those offsets, which is unnecessarily error-prone. A shared rf_test_protocol.h containing the packed structs, frame-layout constants, CRC, and packet parser would substantially reduce the chance of the four firmwares drifting apart.
